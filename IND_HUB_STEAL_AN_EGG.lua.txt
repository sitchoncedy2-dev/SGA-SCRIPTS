local function GetLocalPlotCenter()
	local ok, pt = pcall(function()
		local plot = PlotState and PlotState.ResolvePlot and PlotState.ResolvePlot()
		local cp = plot and plot.CenterPoint
		if typeof(cp) == "Vector3" then return cp end
		if typeof(cp) == "Instance" and cp:IsA("BasePart") then return cp.Position end
		return nil
	end)
	if ok and pt then
		return Vector3.new(pt.X, math.max(pt.Y, 70.4), pt.Z)
	end
	return Vector3.new(464.7, 70.4, -364.0)
end

local function NeutralizeTraps()
	pcall(function()
		local debris = Workspace:FindFirstChild("__DEBRIS")
		if not debris then return end
		for _, d in ipairs(debris:GetChildren()) do
			if d.Name == "PlayerTrap" and d:GetAttribute("Owner") ~= LP.Name then
				if d:IsA("BasePart") then
					d.CanTouch = false
					d.CanQuery = false
				end
				for _, c in ipairs(d:GetDescendants()) do
					if c:IsA("BasePart") then
						c.CanTouch = false
						c.CanQuery = false
						if c.Name == "Hitbox" then
							c.CFrame = CFrame.new(0, -999, 0)
						end
					end
				end
			end
		end
	end)
end

local function MoveTo(target, speed)
	local hrp = findHRP()
	if not hrp or not target then return end
	speed = math.clamp(tonumber(speed) or 450, 50, 750)
	local t0 = os.clock()
	local total = (target - hrp.Position).Magnitude
	while not dead do
		RunService.Heartbeat:Wait()
		hrp = findHRP()
		if not hrp then break end
		local remain = (target - hrp.Position).Magnitude
		if remain < 2 then break end
		local dir = (target - hrp.Position).Unit
		local step = math.min(speed * 0.03, remain)
		local np = hrp.Position + dir * step
		hrp.CFrame = CFrame.lookAt(np, np + dir)
		hrp.AssemblyLinearVelocity = Vector3.zero
		if os.clock() - t0 > total / 35 + 10 then break end
	end
	hrp = findHRP()
	if hrp then
		hrp.CFrame = CFrame.new(target.X, math.max(target.Y, 70), target.Z)
		hrp.AssemblyLinearVelocity = Vector3.zero
	end
end

local function TravelTo(pos, speed)
	local hrp = findHRP()
	if not hrp or not pos then return end
	if avoidTraps then NeutralizeTraps() end
	speed = speed or glideSpeed
	if moveMethod == "Teleport" then
		hrp.CFrame = CFrame.new(pos + Vector3.new(0, 1.5, 0))
		hrp.AssemblyLinearVelocity = Vector3.zero
		return
	end
	local start = hrp.Position
	local safeY = math.max(start.Y, pos.Y, 70.4)
	if moveMethod == "Fly Glide" then
		local alt = safeY + 25
		MoveTo(Vector3.new(start.X, alt, start.Z), speed)
		MoveTo(Vector3.new(pos.X, alt, pos.Z), speed)
		MoveTo(Vector3.new(pos.X, math.max(pos.Y, 70) + 1.2, pos.Z), speed)
		return
	end
	MoveTo(Vector3.new(start.X, safeY, MAIN_ROAD_Z), speed)
	MoveTo(Vector3.new(pos.X, safeY, MAIN_ROAD_Z), speed)
	MoveTo(pos + Vector3.new(0, 1.2, 0), speed)
end

local function isCarrying()
	local pg = LP:FindFirstChildOfClass("PlayerGui")
	local dg = pg and pg:FindFirstChild("DropHeldEgg")
	if dg and dg.Enabled then return true end
	local char = LP.Character
	if char then
		for _, t in ipairs(char:GetChildren()) do
			if t:IsA("Tool") then
				if t:GetAttribute("IsEgg") or t:GetAttribute("Uid") or t:GetAttribute("AssetCategory") then
					return true
				end
				if EggToolDisplay and EggToolDisplay.IsEggTool then
					local ok, r = pcall(EggToolDisplay.IsEggTool, t)
					if ok and r then return true end
				end
			end
		end
	end
	return false
end

local function GetFieldEggs()
	if not EggState or not EggState.ReadFieldEggs then return {} end
	local ok, snap = pcall(EggState.ReadFieldEggs)
	if not ok or not snap or not snap.Records then return {} end
	local list = {}
	for _, rec in ipairs(snap.Records) do
		if rec.State == "Slot" and rec.BoundsCFrame and rec.Uid then
			list[#list + 1] = rec
		end
	end
	return list
end

local function StealEgg(rec)
	if not rec or not rec.Uid or not rec.BoundsCFrame then return false end
	local hrp = findHRP()
	if not hrp then return false end
	local pos = rec.BoundsCFrame.Position
	TravelTo(pos + Vector3.new(0, 1.2, 0), glideSpeed)
	task.wait(0.25)

	local slotKey = nil
	pcall(function()
		if AreaEggSlotIdentity and AreaEggSlotIdentity.LooksLikeFirstAreaUid and AreaEggSlotIdentity.LooksLikeFirstAreaUid(rec.Uid) then
			slotKey = AreaEggSlotIdentity.SlotKey(rec.AreaId, rec.NestId)
		end
	end)

	local carry = GetNetRemote("RF/EggWorld/AskFieldEggCarry")
	if carry then
		pcall(function() carry:InvokeServer({ Uid = rec.Uid, FirstAreaSlotKey = slotKey }) end)
	end
	pcall(function()
		if EggState and EggState.CarryFieldEgg then
			EggState.CarryFieldEgg(rec.Uid, slotKey)
		end
	end)

	pcall(function()
		for _, d in ipairs(Workspace:GetDescendants()) do
			if d:IsA("ProximityPrompt") and d.Name == "CarryAreaEgg" and d.Enabled then
				local p = d.Parent
				if p:IsA("Attachment") then p = p.Parent end
				if p and p:IsA("BasePart") and hrp and (p.Position - hrp.Position).Magnitude < 18 then
					d.HoldDuration = 0
					if fireproximityprompt then fireproximityprompt(d) end
					break
				end
			end
		end
	end)

	local t0 = os.clock()
	while os.clock() - t0 < 1.3 and not dead do
		if isCarrying() then break end
		task.wait(0.05)
	end

	TravelTo(GetLocalPlotCenter() + Vector3.new(0, 1.2, 0), glideSpeed)
	return isCarrying()
end

local function InstantSteal()
	local eggs = GetFieldEggs()
	if #eggs == 0 then
		notify("Steal", "No eggs found")
		return false
	end
	-- prefer priority player plots if we can match area, else first
	local target = eggs[1]
	if #priorityTargets > 0 then
		for _, rec in ipairs(eggs) do
			-- pick first available
			target = rec
			break
		end
	end
	return StealEgg(target)
end

local function StealSlot(slotNum)
	local eggs = GetFieldEggs()
	for _, rec in ipairs(eggs) do
		local nest = tonumber(rec.NestId) or 0
		if nest == slotNum or tostring(rec.NestId) == tostring(slotNum) then
			return StealEgg(rec)
		end
	end
	-- fallback: steal best available
	return InstantSteal()
end

local function SpamAction(n)
	task.spawn(function()
		for i = 1, 8 do
			if n == 1 then
				SelfRagdoll()
			else
				local re = GetNetRemote("RE/BatSwing/Trigger")

			end
			task.wait(0.08)
		end
	end)
end

local function GetClosestPlayer()
	local hrp = findHRP()
	if not hrp then return nil end
	local best, bestDist = nil, math.huge
	for _, p in ipairs(Players:GetPlayers()) do
		if p ~= LP and p.Character then
			local thrp = p.Character:FindFirstChild("HumanoidRootPart")
			if thrp then
				local d = (thrp.Position - hrp.Position).Magnitude
				if d < bestDist then
					bestDist = d
					best = p
				end
			end
		end
	end
	return best
end

local function SpamClosest()
	local t = GetClosestPlayer()
	if not t then
		notify("Spam", "No player nearby")
		return
	end
	notify("Spam", "Target: " .. t.Name)
	-- bat spam near them
	local thrp = t.Character and t.Character:FindFirstChild("HumanoidRootPart")
	if thrp then
		TravelTo(thrp.Position + Vector3.new(0, 1, 0), glideSpeed)
	end
	for _ = 1, 12 do
		local re = GetNetRemote("RE/BatSwing/Trigger")
		if re then pcall(function() re:FireServer() end) end
		task.wait(0.1)
	end
end

local function TPToSlot(slot)
	local plot = Workspace:FindFirstChild("Plots") and Workspace.Plots:FindFirstChild(tostring(slot))
	if plot then
		local pos = plot:FindFirstChild("CenterPoint") and plot.CenterPoint.Position or plot:GetPivot().Position
		TravelTo(pos + Vector3.new(0, 2, 0), glideSpeed)
		notify("Slot", "Traveled to plot " .. tostring(slot))
	else
		notify("Slot", "Plot " .. tostring(slot) .. " not found")
	end
end
