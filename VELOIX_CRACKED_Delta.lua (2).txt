-- ===================== UI HELPERS =====================
local function corner(p, r)
    local c = Instance.new("UICorner")
    c.CornerRadius = UDim.new(0, r or 8)
    c.Parent = p
    return c
end

local function stroke(p, col, t)
    local s = Instance.new("UIStroke")
    s.Color = col or Theme.stroke
    s.Thickness = t or 1
    s.ApplyStrokeMode = Enum.ApplyStrokeMode.Border
    s.Parent = p
    return s
end

local function applyTheme()
    -- recolor main pieces via tags
    for _, v in ipairs(gui:GetDescendants()) do
        if v:GetAttribute("ThemeRole") == "bg" then
            v.BackgroundColor3 = Theme.bg
        elseif v:GetAttribute("ThemeRole") == "panel" then
            v.BackgroundColor3 = Theme.panel
        elseif v:GetAttribute("ThemeRole") == "accent" then
            if v:IsA("TextLabel") or v:IsA("TextButton") then
                v.TextColor3 = Theme.accent
            else
                v.BackgroundColor3 = Theme.accent
            end
        elseif v:GetAttribute("ThemeRole") == "text" then
            v.TextColor3 = Theme.text
        elseif v:GetAttribute("ThemeRole") == "muted" then
            v.TextColor3 = Theme.muted
        elseif v:GetAttribute("ThemeRole") == "btn" then
            if not v:GetAttribute("On") then
                v.BackgroundColor3 = Theme.btn
            end
        elseif v:GetAttribute("ThemeRole") == "stroke" and v:IsA("UIStroke") then
            v.Color = Theme.stroke
        end
    end
end

local function makePage(name)
    local sc = Instance.new("ScrollingFrame")
    sc.Name = name
    sc.Size = UDim2.new(1, 0, 1, 0)
    sc.BackgroundTransparency = 1
    sc.BorderSizePixel = 0
    sc.ScrollBarThickness = 3
    sc.Visible = false
    sc.CanvasSize = UDim2.new(0, 0, 0, 0)
    sc.Parent = content
    local lay = Instance.new("UIListLayout")
    lay.Padding = UDim.new(0, 6)
    lay.Parent = sc
    lay:GetPropertyChangedSignal("AbsoluteContentSize"):Connect(function()
        sc.CanvasSize = UDim2.new(0, 0, 0, lay.AbsoluteContentSize.Y + 16)
    end)
    pages[name] = sc
    return sc
end

local function switchTab(name)
    currentTab = name
    for n, p in pairs(pages) do
        p.Visible = (n == name)
    end
    for _, b in ipairs(tabBar:GetChildren()) do
        if b:IsA("TextButton") then
            local on = b.Name == "Tab_" .. name
            b.BackgroundColor3 = on and Theme.accent or Theme.btn
            b.TextColor3 = on and Color3.new(1, 1, 1) or Theme.text
        end
    end
end
