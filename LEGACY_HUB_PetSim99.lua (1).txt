local function NetFire(name, ...)
	if not Network then return end
	local r = Network:FindFirstChild(name)
	if r and r:IsA("RemoteEvent") then
		pcall(function() r:FireServer(...) end)
	end
end
local function NetInvoke(name, ...)
	if not Network then return end
	local r = Network:FindFirstChild(name)
	if r and r:IsA("RemoteFunction") then
		return pcall(function() return r:InvokeServer(...) end)
	end
	return false
end

-- ===== REAL FARM =====
local function findNearestBreakable()
	local my = hrp()
	if not my then return nil end
	local folder = getBreakables()
	if not folder then return nil end
	local best, bestDist = nil, 1e9
	for _, br in ipairs(folder:GetChildren()) do
		local mesh = br:FindFirstChildWhichIsA("MeshPart") or br:FindFirstChildWhichIsA("BasePart")
		if mesh then
			local d = (mesh.Position - my.Position).Magnitude
			if d < bestDist then
				bestDist = d
				best = br
			end
		end
	end
	return best, bestDist
end

local function farmTick()
	local br, dist = findNearestBreakable()
	if not br then return end
	local my = hrp()
	if my and dist and dist > (S.breakRadius or 80) then
		local mesh = br:FindFirstChildWhichIsA("MeshPart") or br:FindFirstChildWhichIsA("BasePart")
		if mesh then
			pcall(function()
				my.CFrame = CFrame.new(mesh.Position + Vector3.new(0, 5, 0))
			end)
		end
	end
	-- REAL remote
	NetFire("Breakables_PlayerDealDamage", br.Name)
end

-- ===== REAL ORB / LOOTBAG COLLECT =====
local function collectOrbsTick()
	local folder = getOrbs()
	if not folder then return end
	local ids = {}
	for _, orb in ipairs(folder:GetChildren()) do
		local id = tonumber(orb.Name)
		if id then
			table.insert(ids, id)
			pcall(function() orb:Destroy() end)
		end
	end
	if #ids > 0 then
		NetFire("Orbs: Collect", ids)
	end
end

local function collectLootbagsTick()
	local folder = getLootbags()
	if not folder then return end
	local bags = {}
	for _, bag in ipairs(folder:GetChildren()) do
		bags[bag.Name] = bag.Name
		pcall(function() bag:Destroy() end)
	end
	local has = false
	for _ in pairs(bags) do has = true break end
	if has then
		NetFire("Lootbags_Claim", bags)
	end
end

local function collectTick()
	collectOrbsTick()
	collectLootbagsTick()
end

local speedConn
local function applySpeed(on)
	S.speed = on
	if speedConn then pcall(function() speedConn:Disconnect() end) speedConn = nil end
	if on then
		speedConn = track(RunService.Heartbeat:Connect(function()
			if not S.speed then return end
			local h = hum()
			if h then h.WalkSpeed = S.speedVal end
		end))
	else
		local h = hum()
		if h then h.WalkSpeed = 16 end
	end
end

local function applyJump(on)
	S.jump = on
	local h = hum()
	if h then
		pcall(function() h.UseJumpPower = true end)
		h.JumpPower = on and S.jumpVal or 50
	end
end

local flyBV, flyBG, flyConn
local function applyFly(on)
	S.fly = on
	if flyBV then pcall(function() flyBV:Destroy() end) flyBV = nil end
	if flyBG then pcall(function() flyBG:Destroy() end) flyBG = nil end
	if flyConn then pcall(function() flyConn:Disconnect() end) flyConn = nil end
	local r = hrp()
	if not on or not r then return end
	flyBV = Instance.new("BodyVelocity")
	flyBV.MaxForce = Vector3.new(9e9, 9e9, 9e9)
	flyBV.Velocity = Vector3.zero
	flyBV.Parent = r
	flyBG = Instance.new("BodyGyro")
	flyBG.MaxTorque = Vector3.new(9e9, 9e9, 9e9)
	flyBG.P = 2500
	flyBG.Parent = r
	flyConn = track(RunService.RenderStepped:Connect(function()
		if not S.fly or not flyBV then return end
		local cam = Workspace.CurrentCamera
		if not cam then return end
		local d = Vector3.zero
		if UIS:IsKeyDown(Enum.KeyCode.W) then d = d + cam.CFrame.LookVector end
		if UIS:IsKeyDown(Enum.KeyCode.S) then d = d - cam.CFrame.LookVector end
		if UIS:IsKeyDown(Enum.KeyCode.A) then d = d - cam.CFrame.RightVector end
		if UIS:IsKeyDown(Enum.KeyCode.D) then d = d + cam.CFrame.RightVector end
		if UIS:IsKeyDown(Enum.KeyCode.Space) then d = d + Vector3.new(0, 1, 0) end
		if UIS:IsKeyDown(Enum.KeyCode.LeftControl) or UIS:IsKeyDown(Enum.KeyCode.LeftShift) then
			d = d - Vector3.new(0, 1, 0)
		end
		if d.Magnitude > 0 then d = d.Unit * S.flySpd end
		flyBV.Velocity = d
		if flyBG then flyBG.CFrame = cam.CFrame end
	end))
end

	local function toggle(label, default, cb)
		local r = mk("TextButton", {
			Size = UDim2.new(1, -4, 0, 28), BackgroundColor3 = TH.bg, BorderSizePixel = 0,
			Text = "", AutoButtonColor = false, LayoutOrder = #body:GetChildren(),
		}, body)
		round(r, 2)
		mk("TextLabel", {
			Size = UDim2.new(1, -48, 1, 0), Position = UDim2.fromOffset(8, 0),
			BackgroundTransparency = 1, Text = label, Font = Enum.Font.Code, TextSize = 11,
			TextColor3 = TH.text, TextXAlignment = Enum.TextXAlignment.Left,
		}, r)
		local state = default and true or false
		local mark = mk("TextLabel", {
			Size = UDim2.fromOffset(32, 16), Position = UDim2.new(1, -38, 0.5, -8),
			BackgroundColor3 = state and TH.ok or TH.off, BorderSizePixel = 0,
			Text = state and "ON" or "--", Font = Enum.Font.Code, TextSize = 9,
			TextColor3 = state and Color3.new(0, 0, 0) or TH.dim,
		}, r)
		round(mark, 2)
		r.MouseButton1Click:Connect(function()
			state = not state
			mark.BackgroundColor3 = state and TH.ok or TH.off
			mark.Text = state and "ON" or "--"
			mark.TextColor3 = state and Color3.new(0, 0, 0) or TH.dim
			if cb then pcall(cb, state) end
		end)
	end

		local b = mk("TextButton", {
			Size = UDim2.new(1, -4, 0, 28), BackgroundColor3 = TH.bg, BorderSizePixel = 0,

		b.MouseButton1Click:Connect(function() if cb then pcall(cb) end end)
	end

		local hold = mk("Frame", {
			Size = UDim2.new(1, -4, 0, 48), BackgroundColor3 = TH.bg, BorderSizePixel = 0,
			LayoutOrder = #body:GetChildren(),
		}, body)

			Size = UDim2.new(1, -10, 0, 14), Position = UDim2.fromOffset(8, 3),
			BackgroundTransparency = 1, Text = label, Font = Enum.Font.Code, TextSize = 10,
			TextColor3 = TH.dim, TextXAlignment = Enum.TextXAlignment.Left,
		}, hold)
		local box = mk("TextBox", {
			Size = UDim2.new(1, -64, 0, 22), Position = UDim2.fromOffset(8, 20),
			BackgroundColor3 = TH.panel, BorderSizePixel = 0, Text = tostring(def),
			Font = Enum.Font.Code, TextSize = 11, TextColor3 = TH.text, ClearTextOnFocus = false,
		}, hold)
		round(box, 2)
		local go = mk("TextButton", {
			Size = UDim2.fromOffset(46, 22), Position = UDim2.new(1, -52, 0, 20),
			BackgroundColor3 = TH.panel, BorderSizePixel = 0, Text = "SET",
			Font = Enum.Font.Code, TextSize = 10, TextColor3 = TH.accent,
		}, hold)
		round(go, 2)
		local function apply()
			local n = tonumber(box.Text)
			if not n then return end
			n = math.clamp(n, minV, maxV)
			box.Text = tostring(n)
			if cb then pcall(cb, n) end
		end
		go.MouseButton1Click:Connect(apply)
		box.FocusLost:Connect(function(e) if e then apply() end end)
	end

	local dragging, dragStart, startPos
	Head.InputBegan:Connect(function(input)
		if input.UserInputType == Enum.UserInputType.MouseButton1 or input.UserInputType == Enum.UserInputType.Touch then
			dragging = true
			dragStart = input.Position
			startPos = Main.Position
			input.Changed:Connect(function()
				if input.UserInputState == Enum.UserInputState.End then dragging = false end
			end)
		end
	end)
	UIS.InputChanged:Connect(function(input)
		if dragging and (input.UserInputType == Enum.UserInputType.MouseMovement or input.UserInputType == Enum.UserInputType.Touch) then
			local d = input.Position - dragStart
			Main.Position = UDim2.new(startPos.X.Scale, startPos.X.Offset + d.X, startPos.Y.Scale, startPos.Y.Offset + d.Y)
		end
	end)
	UIS.InputBegan:Connect(function(input, gp)
		if gp then return end
		if input.KeyCode == Enum.KeyCode.RightControl then
			Main.Visible = not Main.Visible
		end
	end)
