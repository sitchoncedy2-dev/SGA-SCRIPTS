local function char()
	return LP.Character
end

local function hrp()
	local c = char()
	return c and c:FindFirstChild("HumanoidRootPart")
end

local function hum()
	local c = char()
	return c and c:FindFirstChildOfClass("Humanoid")
end

local function alive(model)
	if not model or not model.Parent then return false end
	local h = model:FindFirstChildOfClass("Humanoid")
	if h then return h.Health > 0 end
	return model:FindFirstChild("HumanoidRootPart") ~= nil or model.PrimaryPart ~= nil
end

local function rootOf(model)
	if not model then return nil end
	return model:FindFirstChild("HumanoidRootPart")
		or model:FindFirstChild("Torso")
		or model:FindFirstChild("UpperTorso")
		or model.PrimaryPart
		or model:FindFirstChildWhichIsA("BasePart")
end

local mobFolders = {
	"Mobs","Mob","Enemies","Enemy","Monsters","Monster","NPCs","NPC",
	"Units","Alive","WorkspaceMonsters","DungeonMobs","ClientMobs"
}

local function gatherMobs()
	local list = {}
	local seen = {}
	local function add(m)
		if not m or seen[m] then return end
		if m:IsA("Model") and alive(m) and m ~= char() then
			if Players:GetPlayerFromCharacter(m) then return end
			seen[m] = true
			list[#list+1] = m
		end
	end

	for _, name in ipairs(mobFolders) do
		local f = workspace:FindFirstChild(name)
		if f then
			for _, ch in ipairs(f:GetChildren()) do
				add(ch)
			end
			for _, d in ipairs(f:GetDescendants()) do
				if d:IsA("Model") and d:FindFirstChildOfClass("Humanoid") then
					add(d)
				end
			end
		end
	end

	-- fallback scan (capped)
	local n = 0
	for _, obj in ipairs(workspace:GetChildren()) do
		if n > 80 then break end
		if obj:IsA("Model") and obj:FindFirstChildOfClass("Humanoid") then
			add(obj)
			n += 1
		end
	end
	return list
end

local function mobScore(m, myPos)
	local r = rootOf(m)
	if not r then return -1e9 end
	local dist = (r.Position - myPos).Magnitude
	local score = -dist
	local name = string.lower(m.Name)
	if cfg.priorityRanged then
		if string.find(name, "mage") or string.find(name, "range") or string.find(name, "archer") or string.find(name, "wizard") then
			score += 40
		end
	end
	if string.find(name, "boss") then
		score += 80
	end
	local h = m:FindFirstChildOfClass("Humanoid")
	if h and h.MaxHealth > 0 then
		score += (1 - h.Health / h.MaxHealth) * 10
	end
	return score
end

local function nearestMob()
	local root = hrp()
	if not root then return nil end
	local myPos = root.Position
	local best, bestS = nil, -1e9
	for _, m in ipairs(gatherMobs()) do
		local s = mobScore(m, myPos)
		if s > bestS then
			bestS = s
			best = m
		end
	end
	return best
end

local function farmOffset(targetRoot)
	local mode = cfg.farmMode
	local h = cfg.farmHeight
	local d = cfg.farmDist
	local look = targetRoot.CFrame.LookVector
	if look.Magnitude < 0.05 then look = Vector3.new(0, 0, -1) else look = look.Unit end
	if mode == "Above" then
		return targetRoot.Position + Vector3.new(0, h, 0)
	elseif mode == "Behind" then
		return targetRoot.Position + look * d + Vector3.new(0, h * 0.35, 0)
	elseif mode == "Front" then
		return targetRoot.Position - look * d + Vector3.new(0, h * 0.35, 0)
	elseif mode == "Orbit" then
		local t = os.clock() * 2.2
		return targetRoot.Position + Vector3.new(math.cos(t) * d, h * 0.5, math.sin(t) * d)
	end
	return targetRoot.Position + Vector3.new(0, h, 0)
end

local function moveTo(pos)
	local root = hrp()
	if not root then return end
	if cfg.tpMode == "Teleport" then
		root.CFrame = CFrame.new(pos)
		root.AssemblyLinearVelocity = Vector3.zero
		return
	end
	local dist = (root.Position - pos).Magnitude
	if dist < 4 then
		root.CFrame = CFrame.new(pos)
		return
	end
	local t = math.clamp(dist / math.max(cfg.tweenSpeed, 40), 0.05, 1.2)
	pcall(function()
		TweenService:Create(root, TweenInfo.new(t, Enum.EasingStyle.Linear), {CFrame = CFrame.new(pos)}):Play()
	end)
end

local function equipWeapon()
	local c = char()
	local bag = LP:FindFirstChild("Backpack")
	if not c or not bag then return end
	local h = hum()
	if not h then return end
	if c:FindFirstChildOfClass("Tool") then return end
	for _, t in ipairs(bag:GetChildren()) do
		if t:IsA("Tool") then
			pcall(function() h:EquipTool(t) end)
			return
		end
	end
end

local attackRemotes = {}
local function cacheAttackRemotes()
	table.clear(attackRemotes)
	local function scan(inst, depth)
		if depth > 5 then return end
		for _, ch in ipairs(inst:GetChildren()) do
			local n = string.lower(ch.Name)
			if ch:IsA("RemoteEvent") or ch:IsA("RemoteFunction") then
				if string.find(n, "attack") or string.find(n, "swing") or string.find(n, "damage")
					or string.find(n, "combat") or string.find(n, "hit") or string.find(n, "ability")
					or string.find(n, "skill") or string.find(n, "cast") or string.find(n, "use") then
					attackRemotes[#attackRemotes+1] = ch
				end
			end
			scan(ch, depth + 1)
		end
	end
	pcall(scan, RS, 0)
end
cacheAttackRemotes()

local function fireAttack()
	equipWeapon()
	local tool = char() and char():FindFirstChildOfClass("Tool")
	if tool then
		pcall(function() tool:Activate() end)
	end
	for _, rem in ipairs(attackRemotes) do
		pcall(function()
			if rem:IsA("RemoteEvent") then
				rem:FireServer()
			else
				rem:InvokeServer()
			end
		end)
	end
	pcall(function()
		VU:CaptureController()
		VU:ClickButton1(Vector2.new(0, 0))
	end)
end

local function pressKey(key)
	pcall(function()
		VIM:SendKeyEvent(true, key, false, game)
		task.wait(0.03)
		VIM:SendKeyEvent(false, key, false, game)
	end)
end

local function tryClickButtons(names)
	local pg = LP:FindFirstChild("PlayerGui")
	if not pg then return end
	for _, g in ipairs(pg:GetDescendants()) do
		if g:IsA("TextButton") or g:IsA("ImageButton") then
			local t = string.lower((g.Text or g.Name or ""))
			for _, n in ipairs(names) do
				if string.find(t, string.lower(n)) and g.Visible and g.AbsoluteSize.X > 0 then
					pcall(function()
						firesignal(g.MouseButton1Click)
					end)
					pcall(function()
						g.MouseButton1Click:Fire()
					end)
				end
			end
		end
	end
end

local function collectNearbyLoot()
	if not cfg.collectLoot then return end
	local root = hrp()
	if not root then return end
	for _, obj in ipairs(workspace:GetDescendants()) do
		if obj:IsA("ProximityPrompt") then
			local pn = string.lower(obj.Name .. (obj.Parent and obj.Parent.Name or ""))
			if string.find(pn, "loot") or string.find(pn, "chest") or string.find(pn, "pickup") or string.find(pn, "item") then
				local part = obj.Parent and (obj.Parent:IsA("BasePart") and obj.Parent or obj.Parent:FindFirstChildWhichIsA("BasePart"))
				if part and (part.Position - root.Position).Magnitude < 40 then
					if fireproximityprompt then
						pcall(fireproximityprompt, obj)
					end
				end
			end
		end
	end
end

-- main farm
task.spawn(function()
	while true do
		task.wait(cfg.hitDelay)
		if not (cfg.autoFarm or cfg.killAura or cfg.autoAttack) then
			task.wait(0.2)
			continue
		end
		local target = nearestMob()
		local root = hrp()
		if not root then continue end

		if target and cfg.autoFarm then
			local tr = rootOf(target)
			if tr then
				moveTo(farmOffset(tr))
				root.CFrame = CFrame.new(root.Position, tr.Position)
			end
		end

		if target and (cfg.autoAttack or cfg.autoFarm or cfg.killAura) then
			local tr = rootOf(target)
			local inRange = true
			if cfg.killAura and tr then
				inRange = (tr.Position - root.Position).Magnitude <= cfg.auraRange
			end
			if inRange or cfg.autoFarm then
				fireAttack()
			end
		end

		if cfg.autoSkill then
			local fighting = target ~= nil
			if not cfg.skillOnlyInCombat or fighting then
				for _, k in ipairs(cfg.skillKeys) do
					pressKey(k)
					task.wait(cfg.skillDelay)
				end
			end
		end

		if cfg.collectLoot then
			collectNearbyLoot()
		end
	end
end)

-- start / replay / lobby
task.spawn(function()
	while true do
		task.wait(1)
		if cfg.autoStart then
			tryClickButtons({"start", "play", "create", "ready", "go"})
		end
		if cfg.autoReplay then
			tryClickButtons({"replay", "again", "restart", "rejoin dungeon"})
			task.wait(cfg.replayDelay)
		end
		if cfg.autoLobby then
			tryClickButtons({"lobby", "leave", "return", "back to lobby"})
		end
	end
end)
