local function safeRequire(...)
	local cur = select(1, ...)
	for i=2, select("#", ...) do
		local name = select(i, ...)
		local ok, n = pcall(function() return cur:WaitForChild(name, 3) end)
		if not ok or not n then return nil end
		cur = n
	end
	local ok, mod = pcall(require, cur)
	return ok and mod or nil
end

local function firePrompt(prompt)
	if not prompt then return end
	pcall(function()
		local old = prompt.HoldDuration
		prompt.HoldDuration = 0
		if fireproximityprompt then
			fireproximityprompt(prompt)
		else
			prompt:InputHoldBegin()
			task.wait(0.05)
			prompt:InputHoldEnd()
		end
		prompt.HoldDuration = old
	end)
end

local function getEggPrompt(model)
	if not model then return nil end
	for _, d in ipairs(model:GetDescendants()) do
		if d:IsA("ProximityPrompt") then return d end
	end
	return nil
end

local function rarityScore(name)
	if not name then return 0 end
	name = tostring(name)
	for k,v in pairs(RarityRank) do
		if name:lower():find(k:lower()) then return v end
	end
	return 0
end

local function minRank()
	return RarityRank[F.minRarity] or 3
end

local function collectEggs()
	local list = {}
	-- EggState snapshot
	if EggState and EggState.ReadFieldEggs then
		local ok, snap = pcall(function() return EggState:ReadFieldEggs() end)
		if ok and type(snap)=="table" then
			for uid, data in pairs(snap) do
				if type(data)=="table" then
					local model = data.model or data.Model
					if not model and data.uid then
						-- try workspace
					end
					local rarity = data.rarity or data.Rarity or (data.record and (data.record.Rarity or data.record.Category)) or "Common"
					local part = model and (model:FindFirstChild("HumanoidRootPart") or model:FindFirstChildWhichIsA("BasePart"))
					if part or model then
						table.insert(list, {
							uid = uid,
							model = model,
							part = part,
							rarity = tostring(rarity),
							score = rarityScore(rarity) * 100000 + ((data.size or data.Size or 1) * 10),
						})
					end
				end
			end
		end
	end
	-- Workspace fallback: prompts named CarryAreaEgg
	for _, d in ipairs(Workspace:GetDescendants()) do
		if d:IsA("ProximityPrompt") and (d.Name=="CarryAreaEgg" or d.Name:lower():find("carry") or d.Name:lower():find("egg")) then
			local model = d:FindFirstAncestorOfClass("Model")
			if model then
				local part = model:FindFirstChildWhichIsA("BasePart")
				local already = false
				for _, e in ipairs(list) do if e.model==model then already=true break end end
				if not already and part then
					local rname = model:GetAttribute("Rarity") or model.Name
					table.insert(list, {
						model = model,
						part = part,
						prompt = d,
						rarity = tostring(rname),
						score = rarityScore(rname) * 100000 + part.Size.Magnitude,
					})
				end
			end
		end
	end
	table.sort(list, function(a,b) return (a.score or 0) > (b.score or 0) end)
	return list
end

local function isCarrying()
	if EggState then
		local ok, r = pcall(function()
			if type(EggState.IsCarrying)=="function" then return EggState:IsCarrying() end
			if EggState.CarryData then return EggState.CarryData.IsCarrying==true end
			return false
		end)
		if ok and r then return true end
	end
	local char = LP.Character
	if not char then return false end
	for _, t in ipairs(char:GetChildren()) do
		if t:IsA("Tool") and (t.Name:lower():find("egg") or t:GetAttribute("IsEgg")) then return true end
	end
	return false
end

local function stealEgg(entry)
	local hrp = getHRP()
	if not hrp or not entry then return false end
	local target = entry.part
	if not target and entry.model then
		target = entry.model:FindFirstChildWhichIsA("BasePart")
	end
	if not target then return false end
	pcall(function()
		hrp.CFrame = target.CFrame + Vector3.new(0, 3, 0)
	end)
	task.wait(0.08)
	local prompt = entry.prompt or getEggPrompt(entry.model)
	if prompt then
		firePrompt(prompt)
		task.wait(F.stealDelay)
		return true
	end
	-- nearby prompts
	for _, d in ipairs(Workspace:GetDescendants()) do
		if d:IsA("ProximityPrompt") and d.Enabled then
			local p = d.Parent
			if p and p:IsA("BasePart") and (p.Position - hrp.Position).Magnitude < 12 then
				firePrompt(d)
				task.wait(F.stealDelay)
				return true
			end
		end
	end
	return false
end

local function findBase()
	local plot = Workspace:FindFirstChild("Plots") or Workspace:FindFirstChild("PlayerPlots")
	if plot then
		local mine = plot:FindFirstChild(LP.Name) or plot:FindFirstChild(tostring(LP.UserId))
		if mine then
			local p = mine:FindFirstChildWhichIsA("BasePart", true)
			if p then return p end
		end
	end
	-- spawn
	local sp = Workspace:FindFirstChild("Spawn") or Workspace:FindFirstChild("Lobby")
	if sp then
		local p = sp:IsA("BasePart") and sp or sp:FindFirstChildWhichIsA("BasePart", true)
		if p then return p end
	end
	return nil
end

local function findTreadmill()
	for _, name in ipairs({"Treadmill", "Treadmills", "Gym", "Speed", "Training"}) do
		local f = Workspace:FindFirstChild(name, true)
		if f then
			local p = f:IsA("BasePart") and f or f:FindFirstChildWhichIsA("BasePart", true)
			if p then return p end
		end
	end
	for _, d in ipairs(Workspace:GetDescendants()) do
		if d:IsA("BasePart") and d.Name:lower():find("tread") then return d end
	end
	return nil
end

local function goTo(part)
	local hrp = getHRP()
	if not hrp or not part then return end
	pcall(function() hrp.CFrame = part.CFrame + Vector3.new(0, 4, 0) end)
end

local function farmTreadmill(seconds)
	local tm = findTreadmill()
	if tm then goTo(tm) end
	local hum = getHum()
	local t0 = os.clock()
	while F.alive and (F.autoFarm or F.autoTreadmill) and os.clock() - t0 < seconds do
		hum = getHum()
		if hum then
			pcall(function()
				hum.WalkSpeed = math.max(hum.WalkSpeed, F.speedVal)
				hum:Move(Vector3.new(1, 0, 0), true)
			end)
		end
		-- walk in place / jump lightly to keep treadmill progress
		local hrp = getHRP()
		if hrp and tm then
			if (hrp.Position - tm.Position).Magnitude > 25 then
				goTo(tm)
			end
		end
		task.wait(0.2)
	end
end

local function stealBest()
	F.status = "Scanning eggs..."
	local eggs = collectEggs()
	local need = minRank()
	for _, e in ipairs(eggs) do
		if rarityScore(e.rarity) >= need then
			F.status = "Stealing " .. tostring(e.rarity)
			if stealEgg(e) then
				task.wait(0.3)
				if isCarrying() then
					F.status = "Returning to base"
					local base = findBase()
					if base then goTo(base) end
					task.wait(0.5)
					-- try place prompts at base
					local hrp = getHRP()
					if hrp then
						for _, d in ipairs(Workspace:GetDescendants()) do
							if d:IsA("ProximityPrompt") and d.Enabled and (d.Parent and d.Parent:IsA("BasePart")) then
								if (d.Parent.Position - hrp.Position).Magnitude < 20 then
									firePrompt(d)
								end
							end
						end
					end
					return true
				end
			end
		end
	end
	-- fallback: any egg
	if #eggs > 0 then
		F.status = "Stealing best available"
		stealEgg(eggs[1])
		task.wait(0.4)
		if isCarrying() then
			local base = findBase()
			if base then goTo(base) end
			return true
		end
	end
	return false
end

local function runAutoFarm()
	while F.alive and F.autoFarm do
		F.status = "Auto Farm: steal phase"
		local stole = stealBest()
		if not F.autoFarm then break end
		F.status = "Auto Farm: treadmill until restock"
		-- farm treadmill while waiting for new eggs (cycle ~20-40s)
		farmTreadmill(25)
		if not F.autoFarm then break end
		F.status = "Auto Farm: checking restock"
		task.wait(1)
	end
	F.status = "Idle"
end

local function setSpeed(on)
	F.speedOn = on
	disconnect(speedConn)
	if on then
		speedConn = RunService.Heartbeat:Connect(function()
			if not F.speedOn then return end
			local h = getHum()
			if h then h.WalkSpeed = F.speedVal end
		end)
		table.insert(featureConns, speedConn)
	else
		local h = getHum()
		if h then h.WalkSpeed = 16 end
	end
end

local function setFly(on)
	F.flyOn = on
	local hrp = getHRP()
	local hum = getHum()
	if flyBV then pcall(function() flyBV:Destroy() end) flyBV=nil end
	if flyBG then pcall(function() flyBG:Destroy() end) flyBG=nil end
	if not on then
		if hum then hum.PlatformStand = false end
		return
	end
	if not hrp or not hum then return end
	hum.PlatformStand = true
	flyBG = Instance.new("BodyGyro")
	flyBG.P = 9e4
	flyBG.MaxTorque = Vector3.new(9e9,9e9,9e9)
	flyBG.CFrame = hrp.CFrame
	flyBG.Parent = hrp
	flyBV = Instance.new("BodyVelocity")
	flyBV.MaxForce = Vector3.new(9e9,9e9,9e9)
	flyBV.Velocity = Vector3.zero
	flyBV.Parent = hrp
	local conn = RunService.RenderStepped:Connect(function()
		if not F.flyOn or not hrp.Parent then disconnect(conn) return end
		local cam = Workspace.CurrentCamera
		local dir = Vector3.zero
		if UserInputService:IsKeyDown(Enum.KeyCode.W) then dir=dir+cam.CFrame.LookVector end
		if UserInputService:IsKeyDown(Enum.KeyCode.S) then dir=dir-cam.CFrame.LookVector end
		if UserInputService:IsKeyDown(Enum.KeyCode.A) then dir=dir-cam.CFrame.RightVector end
		if UserInputService:IsKeyDown(Enum.KeyCode.D) then dir=dir+cam.CFrame.RightVector end
		if UserInputService:IsKeyDown(Enum.KeyCode.Space) then dir=dir+Vector3.new(0,1,0) end
		if UserInputService:IsKeyDown(Enum.KeyCode.LeftControl) then dir=dir-Vector3.new(0,1,0) end
		if dir.Magnitude>0 then dir=dir.Unit*F.flySpeed end
		flyBV.Velocity = dir
		flyBG.CFrame = cam.CFrame
	end)
	table.insert(featureConns, conn)
end

local function slider(page, label, order, min, max, default, cb)
	local row = Instance.new("Frame")
	row.Size = UDim2.new(1,-8,0,44)
	row.BackgroundColor3 = Color3.fromRGB(18,20,38)
	row.BorderSizePixel = 0
	row.LayoutOrder = order
	row.Parent = page
	corner(row, 6)
	local val = default
	local t = Instance.new("TextLabel")
	t.Size = UDim2.new(1,-12,0,18)
	t.Position = UDim2.fromOffset(10,4)
	t.BackgroundTransparency = 1
	t.Text = label..": "..tostring(val)
	t.TextColor3 = COLORS.text
	t.Font = Enum.Font.Gotham
	t.TextSize = 12
	t.TextXAlignment = Enum.TextXAlignment.Left
	t.Parent = row
	local bar = Instance.new("Frame")
	bar.Size = UDim2.new(1,-20,0,8)
	bar.Position = UDim2.fromOffset(10,28)
	bar.BackgroundColor3 = Color3.fromRGB(30,35,60)
	bar.BorderSizePixel = 0
	bar.Parent = row
	corner(bar, 4)
	local fill = Instance.new("Frame")
	fill.Size = UDim2.new((val-min)/(max-min),0,1,0)
	fill.BackgroundColor3 = COLORS.accent
	fill.BorderSizePixel = 0
	fill.Parent = bar
	corner(fill, 4)
	local dragging = false
	local function update(x)
		local rel = math.clamp((x-bar.AbsolutePosition.X)/math.max(bar.AbsoluteSize.X,1),0,1)
		val = math.floor(min+(max-min)*rel+0.5)
		fill.Size = UDim2.new(rel,0,1,0)
		t.Text = label..": "..tostring(val)
		if cb then cb(val) end
	end
	bar.InputBegan:Connect(function(input)
		if input.UserInputType==Enum.UserInputType.MouseButton1 or input.UserInputType==Enum.UserInputType.Touch then
			dragging=true update(input.Position.X)
		end
	end)
	UserInputService.InputEnded:Connect(function(input)
		if input.UserInputType==Enum.UserInputType.MouseButton1 or input.UserInputType==Enum.UserInputType.Touch then dragging=false end
	end)
	UserInputService.InputChanged:Connect(function(input)
		if dragging and (input.UserInputType==Enum.UserInputType.MouseMovement or input.UserInputType==Enum.UserInputType.Touch) then update(input.Position.X) end
	end)
end
