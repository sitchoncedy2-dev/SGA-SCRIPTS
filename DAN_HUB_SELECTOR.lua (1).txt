local getgenv = getgenv or function()
    return _G
end

if getgenv().DAN_HUB_SELECTOR_LOADED then
    warn("DAN HUB SELECTOR already loaded")
    return
end
getgenv().DAN_HUB_SELECTOR_LOADED = true

local WindUI = loadstring(game:HttpGet("https://github.com/Footagesus/WindUI/releases/latest/download/main.lua"))()

local Window = WindUI:CreateWindow({
    Title = "DAN HUB SELECTOR",
    Author = "Made by SGA / SUN GOD NIKA",
    Folder = "DanHubSelector",
    Size = UDim2.fromOffset(480, 360),
    Transparent = true,
    Theme = "Dark"
})

local ScriptsTab = Window:Tab({Title = "DAN HUB", Icon = "zap"})
local CreditsTab = Window:Tab({Title = "Credits", Icon = "info"})

local function loadUrl(name, url)
    WindUI:Notify({
        Title = "DAN HUB",
        Content = "Loading " .. name .. "...",
        Duration = 3
    })
    task.spawn(function()
        local ok, err = pcall(function()
            local src = game:HttpGet(url, true)
            if type(src) ~= "string" or #src < 10 then
                error("Empty response")
            end
            local fn, loadErr = loadstring(src)
            if not fn then
                error(tostring(loadErr))
            end
            fn()
        end)
        if ok then
            WindUI:Notify({
                Title = "DAN HUB",
                Content = name .. " loaded!",
                Duration = 3
            })
        else
            WindUI:Notify({
                Title = "DAN HUB",
                Content = "Failed: " .. tostring(err),
                Duration = 5
            })
            warn("[DAN HUB SELECTOR] " .. tostring(err))
        end
    end)
end
