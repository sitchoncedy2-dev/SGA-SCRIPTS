local function createAfkDisplay(character)
    if not character then
        return
    end
    local head = character:WaitForChild("Head", 5)
    if not head or head:FindFirstChild("AFK_Tag") then
        return
    end
    local billboard = Instance.new("BillboardGui")
    billboard.Name = "AFK_Tag"
    billboard.Size = UDim2.new(0, 170, 0, 28)
    billboard.StudsOffset = Vector3.new(0, 3, 0)
    billboard.AlwaysOnTop = true
    billboard.Parent = head
    local label = Instance.new("TextLabel")
    label.Size = UDim2.new(1, 0, 1, 0)
    label.BackgroundTransparency = 1
    label.Text = "DAN HUB ANTI-AFK [00:00:00]"
    label.TextColor3 = Color3.fromRGB(0, 255, 200)
    label.Font = Enum.Font.GothamBold
    label.TextSize = 12
    label.TextStrokeTransparency = 0.2
    label.Parent = billboard
    local stroke = Instance.new("UIStroke")
    stroke.Thickness = 1.4
    stroke.Color = Color3.fromRGB(0, 0, 0)
    stroke.Parent = label
    task.spawn(function()
        while billboard and billboard.Parent do
            local e = os.time() - startTime
            local h = math.floor(e / 3600)
            local m = math.floor((e % 3600) / 60)
            local s = e % 60
            label.Text = string.format("DAN HUB ANTI-AFK [%02d:%02d:%02d]", h, m, s)
            task.wait(1)
        end
    end)
end

local function getEquippedTool()
    local char = LocalPlayer.Character
    if not char then
        return
    end
    local tool = char:FindFirstChildOfClass("Tool")
    if tool and not string.find(string.lower(tool.Name), "punch") then
        return tool
    end
    local backpack = LocalPlayer:FindFirstChild("Backpack")
    if backpack then
        for _, item in ipairs(backpack:GetChildren()) do
            if item:IsA("Tool") and not string.find(string.lower(item.Name), "punch") then
                char.Humanoid:EquipTool(item)
                return item
            end
        end
    end
    return nil
end

local function equipPunch()
    local char = LocalPlayer.Character
    local backpack = LocalPlayer:FindFirstChild("Backpack")
    if not char then
        return nil
    end
    local punch = char:FindFirstChild("Punch")
    if punch then
        return punch
    end
    if backpack then
        punch = backpack:FindFirstChild("Punch")
        if punch then
            char.Humanoid:EquipTool(punch)
            return punch
        end
    end
    return nil
end

local function massTogglePets(rarity, state)
    pcall(function()
        local equipRemote = RemoteEvents:FindFirstChild("equipPetEvent")
        local petsFolder = LocalPlayer:FindFirstChild("petsFolder")

        if equipRemote and rarityFolder then
            for _, pet in ipairs(rarityFolder:GetChildren()) do
                equipRemote:FireServer(state and "equipPet" or "unequipPet", pet)
                task.wait(0.02)
            end
        end
    end)
end

local function getRock(rockName)
    local folder = workspace:FindFirstChild("machinesFolder")
    if not folder then
        return nil
    end
    local rock = folder:FindFirstChild(rockName)
    if rock then
        return rock
    end
    for _, obj in ipairs(folder:GetChildren()) do
        if string.find(string.lower(obj.Name), string.lower(rockName)) then
            return obj
        end
    end
    return nil
end

local function getRockCFrame(rock)
    if not rock then
        return nil
    end
    if rock:IsA("Model") then
        local primary = rock.PrimaryPart or rock:FindFirstChildWhichIsA("BasePart")
        if primary then
            return primary.CFrame
        end
    elseif rock:IsA("BasePart") then
        return rock.CFrame
    end
    for _, part in ipairs(rock:GetDescendants()) do
        if part:IsA("BasePart") then
            return part.CFrame
        end
    end
    return nil
end

local function punchRockLoop(flagName, rockName)
    task.spawn(function()
        while getgenv()[flagName] do
            pcall(function()
                local rock = getRock(rockName)
                local rockCF = getRockCFrame(rock)
                local char = LocalPlayer.Character
                local root = char and char:FindFirstChild("HumanoidRootPart")
                if rockCF and root then
                    root.CFrame = rockCF * CFrame.new(0, 0, 4)
                end
                local punch = equipPunch()
                if punch then
                    punch:Activate()
                end
                if MuscleEvent then
                    MuscleEvent:FireServer("punch")
                    MuscleEvent:FireServer("rep")
                end
            end)
            task.wait(0.12)
        end
    end)
end

local function readLocalStats()
    local stats = {
        Strength = 0,
        Durability = 0,
        Rebirths = 0,
        Kills = 0,
        Brawls = 0
    }
    pcall(function()
        local ls = LocalPlayer:FindFirstChild("leaderstats")
        if ls then
            for _, v in ipairs(ls:GetChildren()) do
                local n = string.lower(v.Name)
                local val = tonumber(v.Value) or 0
                if string.find(n, "stren") then
                    stats.Strength = val
                elseif string.find(n, "dura") then
                    stats.Durability = val
                elseif string.find(n, "rebirth") then
                    stats.Rebirths = val
                elseif string.find(n, "kill") then
                    stats.Kills = val
                elseif string.find(n, "brawl") then
                    stats.Brawls = val
                end
            end
        end
        -- Muscle Legends often stores more under player folders
        for _, folderName in ipairs({"MuscleStats", "stats", "Stats", "playerStats"}) do
            local f = LocalPlayer:FindFirstChild(folderName)
            if f then
                for _, v in ipairs(f:GetDescendants()) do
                    if v:IsA("NumberValue") or v:IsA("IntValue") then
                        local n = string.lower(v.Name)
                        local val = tonumber(v.Value) or 0
                        if string.find(n, "stren") and val > stats.Strength then
                            stats.Strength = val
                        elseif string.find(n, "dura") and val > stats.Durability then
                            stats.Durability = val
                        elseif string.find(n, "rebirth") and val > stats.Rebirths then
                            stats.Rebirths = val
                        end
                    end
                end
            end
        end
    end)
    return stats
end

local function formatNum(n)
    n = tonumber(n) or 0
    local abs = math.abs(n)
    if abs >= 1e12 then
        return string.format("%.2fT", n / 1e12)
    elseif abs >= 1e9 then
        return string.format("%.2fB", n / 1e9)
    elseif abs >= 1e6 then
        return string.format("%.2fM", n / 1e6)
    elseif abs >= 1e3 then
        return string.format("%.2fK", n / 1e3)
    end
    return string.format("%.0f", n)
end

local statStart = readLocalStats()
local statStartTime = os.clock()
local lastStatSnapshot = {
    Strength = statStart.Strength,
    Durability = statStart.Durability
}

local function ratesFor(delta, seconds)
    if seconds <= 0 then
        return 0, 0, 0, 0, 0, 0
    end
    local perSec = delta / seconds
    return perSec * 60, perSec * 3600, perSec * 86400, perSec * 604800, perSec * 2592000, perSec * 31536000
end

local function readPlayerStats(player)
    local stats = {}
    if not player then
        return stats
    end
    pcall(function()
        local ls = player:FindFirstChild("leaderstats")
        if ls then
            for _, v in ipairs(ls:GetChildren()) do
                if v:IsA("NumberValue") or v:IsA("IntValue") or v:IsA("StringValue") then
                    stats[v.Name] = v.Value
                end
            end
        end
        for _, folderName in ipairs({"MuscleStats", "stats", "Stats", "playerStats"}) do
            local f = player:FindFirstChild(folderName)
            if f then
                for _, v in ipairs(f:GetDescendants()) do
                    if v:IsA("NumberValue") or v:IsA("IntValue") then
                        stats[v.Name] = v.Value
                    end
                end
            end
        end
    end)
    return stats
end

local function playerList()
    local names = {}
    for _, p in ipairs(Players:GetPlayers()) do
        if p ~= LocalPlayer then
            names[#names + 1] = p.Name
        end
    end
    table.sort(names)
    if #names == 0 then
        names[1] = "(no other players)"
    end
    return names
end

local function punchTarget(plr)
    if not plr or not plr.Character then
        return
    end
    local targetRoot = plr.Character:FindFirstChild("HumanoidRootPart")
    local myChar = LocalPlayer.Character
    local myRoot = myChar and myChar:FindFirstChild("HumanoidRootPart")
    if not targetRoot or not myRoot then
        return
    end
    myRoot.CFrame = targetRoot.CFrame * CFrame.new(0, 0, 3)
    local punch = equipPunch()
    if punch then
        punch:Activate()
    end
    if MuscleEvent then
        MuscleEvent:FireServer("punch")
        MuscleEvent:FireServer("rep")
    end
end

local function isWhitelisted(name)
    if not name then
        return false
    end
    for _, n in ipairs(getgenv().whitelistNames) do
        if n == name then
            return true
        end
    end
    return false
end
