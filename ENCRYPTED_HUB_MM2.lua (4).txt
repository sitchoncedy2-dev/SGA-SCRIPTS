local function getChar() return LP.Character end
local function getHRP()
	local c = getChar()
	return c and c:FindFirstChild("HumanoidRootPart")
end
local function getHum()
	local c = getChar()
	return c and c:FindFirstChildOfClass("Humanoid")
end
local function getRoot(plr)
	local c = plr and plr.Character
	return c and c:FindFirstChild("HumanoidRootPart")
end

local function getRole(plr)
	if not plr then return "Unknown" end
	local ok, attr = pcall(function() return plr:GetAttribute("Role") or plr:GetAttribute("role") end)
	if ok and attr and tostring(attr) ~= "" then return tostring(attr) end
	local function hasTool(name)
		name = name:lower()
		local function scan(container)
			if not container then return false end
			for _, t in ipairs(container:GetChildren()) do
				if t:IsA("Tool") and t.Name:lower():find(name, 1, true) then return true end
			end
			return false
		end
		return scan(plr.Character) or scan(plr:FindFirstChildOfClass("Backpack"))
	end
	if hasTool("knife") then return "Murderer" end
	if hasTool("gun") then return "Sheriff" end
	return "Innocent"
end

local function isMurderer(plr)
	return tostring(getRole(plr)):lower():find("murder") ~= nil
end
local function isSheriff(plr)
	local r = tostring(getRole(plr)):lower()
	return r:find("sheriff") ~= nil or r:find("hero") ~= nil
end

local function equipTool(sub)
	sub = sub:lower()
	local char = LP.Character
	local pack = LP:FindFirstChildOfClass("Backpack")
	local function try(container)
		if not container then return nil end
		for _, t in ipairs(container:GetChildren()) do
			if t:IsA("Tool") and t.Name:lower():find(sub, 1, true) then
				pcall(function()
					local h = getHum()
					if h then h:EquipTool(t) end
				end)
				return t
			end
		end
		return nil
	end
	return try(char) or try(pack)
end

local function activateTool()
	local char = LP.Character
	if not char then return end
	local tool = char:FindFirstChildOfClass("Tool")
	if tool then pcall(function() tool:Activate() end) end
end

-- anti-hit (lightweight, safe)
local AH = { fake = nil, step = nil, original = nil }
local function stopAntiHit()
	S.antiHit = false
	if AH.step then disc(AH.step) AH.step = nil end
	if AH.fake then pcall(function() AH.fake:Destroy() end) AH.fake = nil end
	local hrp = getHRP()
	if hrp and AH.original then pcall(function() hrp.CFrame = AH.original end) end
end
local function startAntiHit()
	S.antiHit = true
	local hrp = getHRP()
	if not hrp then return end
	AH.original = hrp.CFrame
	if AH.step then disc(AH.step) end
	AH.step = bind(RunService.Heartbeat:Connect(function()
		if not S.antiHit then return end
		local root = getHRP()
		if not root then return end
		-- soft desync offset
		pcall(function()
			root.CFrame = root.CFrame + Vector3.new(0, 0.01, 0)
			root.CFrame = root.CFrame - Vector3.new(0, 0.01, 0)
		end)
	end))
end

-- movement
local flyBG, flyBV, noclipC, speedC
local function setSpeed(on)
	S.speedOn = on
	if speedC then disc(speedC) speedC = nil end
	if on then
		speedC = bind(RunService.Heartbeat:Connect(function()
			if not S.speedOn then return end
			local h = getHum()
			if h then h.WalkSpeed = S.speed end
		end))
	end
end
local function setJump(on)
	S.jumpOn = on
	bind(RunService.Heartbeat:Connect(function()
		if not S.jumpOn then return end
		local h = getHum()
		if h then h.JumpPower = S.jump end
	end))
end
local function setFly(on)
	S.fly = on
	if flyBG then pcall(function() flyBG:Destroy() end) flyBG = nil end
	if flyBV then pcall(function() flyBV:Destroy() end) flyBV = nil end
	local hrp, hum = getHRP(), getHum()
	if not on or not hrp or not hum then return end
	hum.PlatformStand = true
	flyBG = Instance.new("BodyGyro")
	flyBG.P = 9e4
	flyBG.MaxTorque = Vector3.new(9e9, 9e9, 9e9)
	flyBG.CFrame = hrp.CFrame
	flyBG.Parent = hrp
	flyBV = Instance.new("BodyVelocity")
	flyBV.MaxForce = Vector3.new(9e9, 9e9, 9e9)
	flyBV.Velocity = Vector3.zero
	flyBV.Parent = hrp
	bind(RunService.RenderStepped:Connect(function()
		if not S.fly or not hrp or not hrp.Parent then return end
		local cam = Workspace.CurrentCamera
		if not cam then return end
		local dir = Vector3.zero
		if UserInputService:IsKeyDown(Enum.KeyCode.W) then dir = dir + cam.CFrame.LookVector end
		if UserInputService:IsKeyDown(Enum.KeyCode.S) then dir = dir - cam.CFrame.LookVector end
		if UserInputService:IsKeyDown(Enum.KeyCode.A) then dir = dir - cam.CFrame.RightVector end
		if UserInputService:IsKeyDown(Enum.KeyCode.D) then dir = dir + cam.CFrame.RightVector end
		if UserInputService:IsKeyDown(Enum.KeyCode.Space) then dir = dir + Vector3.new(0,1,0) end
		if UserInputService:IsKeyDown(Enum.KeyCode.LeftControl) then dir = dir - Vector3.new(0,1,0) end
		if dir.Magnitude > 0 then dir = dir.Unit * S.flySpeed end
		flyBV.Velocity = dir
		flyBG.CFrame = cam.CFrame
	end))
end

-- main loops
task.spawn(function()
	while S.alive do
		task.wait(0.12)
		local my = getHRP()
		local hum = getHum()

		if S.killAura and my then
			equipTool("knife")
			for _, plr in ipairs(Players:GetPlayers()) do
				if plr ~= LP and plr.Character then
					local r = plr.Character:FindFirstChild("HumanoidRootPart")
					local h = plr.Character:FindFirstChildOfClass("Humanoid")
					if r and h and h.Health > 0 and (r.Position - my.Position).Magnitude <= S.killRange then
						activateTool()
						break
					end
				end
			end
		end
		if S.autoKnife then equipTool("knife") activateTool() end
		if S.autoShoot and my then
			for _, plr in ipairs(Players:GetPlayers()) do
				if plr ~= LP and isMurderer(plr) and plr.Character then
					local r = plr.Character:FindFirstChild("HumanoidRootPart")
					if r and (r.Position - my.Position).Magnitude <= S.shootRange then
						equipTool("gun")
						activateTool()
						break
					end
				end
			end
		end
		if S.autoGun then equipTool("gun") end
		if S.coinFarm and my then
			for _, d in ipairs(Workspace:GetDescendants()) do
				if d:IsA("BasePart") and d.Name:lower():find("coin") then
					pcall(function() my.CFrame = d.CFrame + Vector3.new(0, 2, 0) end)
					break
				end
			end
		end
		if S.antiStun and hum then
			pcall(function()
				hum.PlatformStand = false
				hum:SetStateEnabled(Enum.HumanoidStateType.Ragdoll, false)
				hum:SetStateEnabled(Enum.HumanoidStateType.FallingDown, false)
			end)
		end
		if S.fullbright then
			pcall(function()
				Lighting.Brightness = 2
				Lighting.ClockTime = 14
				Lighting.FogEnd = 1e6
			end)
		end
		if S.noFog then pcall(function() Lighting.FogEnd = 1e6 end) end
		if S.antiAFK then
			pcall(function()
				VirtualUser:CaptureController()
				VirtualUser:ClickButton2(Vector2.new())
			end)
		end

		-- flag-driven extras
		local f = S.flags
		if f["m_aura"] and my then
			equipTool("knife")
			local range = tonumber(f["m_aura_range"]) or 12
			for _, plr in ipairs(Players:GetPlayers()) do
				if plr ~= LP and plr.Character then
					local r = plr.Character:FindFirstChild("HumanoidRootPart")
					if r and (r.Position - my.Position).Magnitude <= range then activateTool() end
				end
			end
		end
		if f["m_chase"] and my then
			local best, bd = nil, 80
			for _, plr in ipairs(Players:GetPlayers()) do
				if plr ~= LP and plr.Character then
					local r = plr.Character:FindFirstChild("HumanoidRootPart")
					if r then
						local d = (r.Position - my.Position).Magnitude
						if d < bd then best, bd = r, d end
					end
				end
			end
			if best then
				pcall(function() my.CFrame = my.CFrame:Lerp(CFrame.new(best.Position + Vector3.new(0,0,3)), 0.2) end)
				equipTool("knife") activateTool()
			end
		end
		if f["s_track"] and my then
			for _, plr in ipairs(Players:GetPlayers()) do
				if plr ~= LP and isMurderer(plr) and plr.Character then
					local r = plr.Character:FindFirstChild("HumanoidRootPart")
					if r then
						pcall(function() my.CFrame = CFrame.new(my.Position, r.Position) end)
						if f["s_shoot"] then equipTool("gun") activateTool() end
					end
				end
			end
		end
		if f["i_run"] and my then
			for _, plr in ipairs(Players:GetPlayers()) do
				if plr ~= LP and isMurderer(plr) and plr.Character then
					local r = plr.Character:FindFirstChild("HumanoidRootPart")
					if r and (r.Position - my.Position).Magnitude < 40 then
						local away = (my.Position - r.Position)
						if away.Magnitude > 0.1 then
							away = away.Unit * 12
							pcall(function() my.CFrame = CFrame.new(my.Position + away) end)
						end
					end
				end
			end
		end
		if f["coin_magnet"] and my then
			for _, d in ipairs(Workspace:GetDescendants()) do
				if d:IsA("BasePart") and d.Name:lower():find("coin") then
					pcall(function() d.CFrame = my.CFrame end)
				end
			end
		end
		if hum then
			for spd = 16, 60 do
				if f["spd_" .. spd] then hum.WalkSpeed = spd end
			end
			for jp = 50, 150, 5 do
				if f["jp_" .. jp] then hum.JumpPower = jp end
			end
		end
		if f["spin"] and my then
			my.CFrame = my.CFrame * CFrame.Angles(0, math.rad(12), 0)
		end
	end
end)

local dragging, dragStart, startPos
Header.InputBegan:Connect(function(input)
	if input.UserInputType == Enum.UserInputType.MouseButton1 or input.UserInputType == Enum.UserInputType.Touch then
		dragging = true
		dragStart = input.Position
		startPos = Main.Position
		input.Changed:Connect(function()
			if input.UserInputState == Enum.UserInputState.End then dragging = false end
		end)
	end
end)
UserInputService.InputChanged:Connect(function(input)
	if dragging and (input.UserInputType == Enum.UserInputType.MouseMovement or input.UserInputType == Enum.UserInputType.Touch) then
		local delta = input.Position - dragStart
		Main.Position = UDim2.new(startPos.X.Scale, startPos.X.Offset + delta.X, startPos.Y.Scale, startPos.Y.Offset + delta.Y)
	end
end)
