local function isProteinOrConsumable(name)
    local n = string.lower(tostring(name or ""))
    return string.find(n, "protein")
        or string.find(n, "shake")
        or string.find(n, "potion")
        or string.find(n, "boost")
        or string.find(n, "food")
        or string.find(n, "drink")
end

local function getEquippedTool()
    local char = LocalPlayer.Character
    if not char then
        return
    end
    local tool = char:FindFirstChildOfClass("Tool")
    if tool then
        local n = string.lower(tool.Name)
        if not string.find(n, "punch") and not isProteinOrConsumable(tool.Name) then
            return tool
        end
    end
    local backpack = LocalPlayer:FindFirstChild("Backpack")
    if backpack then
        for _, item in ipairs(backpack:GetChildren()) do
            if item:IsA("Tool") then
                local n = string.lower(item.Name)
                if not string.find(n, "punch") and not isProteinOrConsumable(item.Name) then
                    local hum = char:FindFirstChildOfClass("Humanoid")
                    if hum then
                        hum:EquipTool(item)
                    end
                    return item
                end
            end
        end
    end
    return nil
end

local function unequipProtein()
    pcall(function()
        local char = LocalPlayer.Character
        if not char then
            return
        end
        local tool = char:FindFirstChildOfClass("Tool")
        if tool and isProteinOrConsumable(tool.Name) then
            local hum = char:FindFirstChildOfClass("Humanoid")
            if hum then
                hum:UnequipTools()
            end
        end
    end)
end

local function equipPunch()
    local char = LocalPlayer.Character
    local backpack = LocalPlayer:FindFirstChild("Backpack")
    if not char then

    local punch = char:FindFirstChild("Punch")
    if punch then
        return punch
    end
    if backpack then
        punch = backpack:FindFirstChild("Punch")
        if punch then
            char.Humanoid:EquipTool(punch)
            return punch
        end
    end
    return nil
end

local function getRock(rockName)
    local folder = workspace:FindFirstChild("machinesFolder")
    if not folder then
        return nil
    end
    local rock = folder:FindFirstChild(rockName)
    if rock then
        return rock
    end
    for _, obj in ipairs(folder:GetChildren()) do
        if string.find(string.lower(obj.Name), string.lower(rockName)) then
            return obj
        end
    end
    return nil
end

local function getRockCFrame(rock)
    if not rock then
        return nil
    end
    if rock:IsA("Model") then
        local primary = rock.PrimaryPart or rock:FindFirstChildWhichIsA("BasePart")
        if primary then
            return primary.CFrame
        end
    elseif rock:IsA("BasePart") then
        return rock.CFrame
    end
    for _, part in ipairs(rock:GetDescendants()) do
        if part:IsA("BasePart") then
            return part.CFrame
        end
    end
    return nil
end

                end
            end)
            task.wait(0.12)
        end
    end)
end

local function equipExerciseTool(toolName)
    local char = LocalPlayer.Character
    local backpack = LocalPlayer:FindFirstChild("Backpack")
    local hum = char and char:FindFirstChildOfClass("Humanoid")
    if not char or not hum then
        return false
    end
    if char:FindFirstChild(toolName) then
        return true
    end
    if backpack then
        local tool = backpack:FindFirstChild(toolName)
        if tool then
            hum:EquipTool(tool)
            return true
        end
        -- giveTool fallback
        local give = RemoteEvents and RemoteEvents:FindFirstChild("giveTool")
        if give then
            pcall(function()
                give:FireServer(toolName)
            end)
            task.wait(0.1)
            tool = backpack:FindFirstChild(toolName) or char:FindFirstChild(toolName)
            if tool and tool.Parent == backpack then
                hum:EquipTool(tool)
            end
            return tool ~= nil
        end
    end
    return false
end

local function doExerciseRep()
    local player = LocalPlayer
    local toolName = getgenv().selectedExercise or "Weight"
    local ev = player:FindFirstChild("muscleEvent") or MuscleEvent
    if toolName == "Fast Punch" then
        equipExerciseTool("Punch")
        if ev then
            ev:FireServer("punch", "rightHand")
            ev:FireServer("punch", "leftHand")
        end
        local punch = player.Character and player.Character:FindFirstChild("Punch")
        if punch then
            punch:Activate()
        end
    elseif toolName == "Stomp" then
        equipExerciseTool("Stomp")
        if ev then
            ev:FireServer("stomp")
        end
        local t = player.Character and player.Character:FindFirstChild("Stomp")
        if t then
            t:Activate()
        end
    elseif toolName == "Ground Slam" then
        equipExerciseTool("Ground Slam")
        if ev then
            ev:FireServer("slam")
        end
        local t = player.Character and player.Character:FindFirstChild("Ground Slam")
        if t then
            t:Activate()
        end
    else
        -- Weight / Pushups / Situps / Handstands
        equipExerciseTool(toolName)
        if ev then
            ev:FireServer("rep")
        end
    end
end

local function getNearestMachineCFrame()
    local folder = workspace:FindFirstChild("machinesFolder")
    local myChar = LocalPlayer.Character
    local myRoot = myChar and myChar:FindFirstChild("HumanoidRootPart")
    if not folder or not myRoot then
        return nil
    end
    local best, bestDist = nil, math.huge
    for _, obj in ipairs(folder:GetDescendants()) do
        if obj:IsA("BasePart") then
            local d = (obj.Position - myRoot.Position).Magnitude
            if d < bestDist then
                bestDist = d
                best = obj
            end
        end
    end
    if best then
        return best.CFrame * CFrame.new(0, 0, 3)
    end
    return nil
end

pcall(function()
    local folders = {}

    local runtime = shared and shared:FindFirstChild("runtime")
    if runtime and runtime:FindFirstChild("cPetShopFolder") then

        table.insert(folders, ReplicatedStorage.cPetShopFolder)
    end
    local seen = {}
    for _, n in ipairs(allPets) do
        seen[n] = true
    end
    for _, folder in ipairs(folders) do
        for _, child in ipairs(folder:GetChildren()) do
            if not seen[child.Name] then
                allPets[#allPets + 1] = child.Name
                seen[child.Name] = true
            end
        end
    end
    table.sort(allPets)
end)

local function buyPetShop(petName)
    if not petName then
        return false
    end
    local ok = false
    pcall(function()
        local remote = ReplicatedStorage:FindFirstChild("rEvents")
            and ReplicatedStorage.rEvents:FindFirstChild("cPetShopRemote")
        if not remote then
            remote = ReplicatedStorage:FindFirstChild("cPetShopRemote")
        end
        local petObj = nil
        local shared = ReplicatedStorage:FindFirstChild("shared")
        local runtime = shared and shared:FindFirstChild("runtime")
        local folder = runtime and runtime:FindFirstChild("cPetShopFolder")
        if folder then
            petObj = folder:FindFirstChild(petName)
        end
        if not petObj then
            folder = ReplicatedStorage:FindFirstChild("cPetShopFolder")
            petObj = folder and folder:FindFirstChild(petName)
        end
        if remote and petObj then
            if remote:IsA("RemoteFunction") then
                remote:InvokeServer(petObj)
            else
                remote:FireServer(petObj)
            end
            ok = true
            return
        end
        -- fallback: DAN public style root remote + folder
        if not remote then
            remote = ReplicatedStorage:FindFirstChild("cPetShopRemote")
        end
        if remote and petObj and remote:IsA("RemoteFunction") then
            remote:InvokeServer(petObj)
            ok = true
        end
    end)
    return ok
end

-- keep old multi-try path disabled; single correct InvokeServer only
local function buyPetShop_DISABLED_OLD(petName)
    if not petName then
        return false
    end
    local ok = false
    pcall(function()
        local folder = ReplicatedStorage:FindFirstChild("cPetShopFolder") or PetShopFolder
        local remote = ReplicatedStorage:FindFirstChild("cPetShopRemote") or PetShopRemote
        if not remote then
            return
        end
        local petObj = folder and (folder:FindFirstChild(petName) or folder:FindFirstChild(petName, true))
        if remote:IsA("RemoteFunction") then
            if petObj then
                pcall(function() remote:InvokeServer(petObj) end)
            end
            ok = true
        elseif remote:IsA("RemoteEvent") then
            if petObj then
                pcall(function() remote:FireServer(petObj) end)
                pcall(function() remote:FireServer("buy", petObj) end)
            end
            pcall(function() remote:FireServer(petName) end)
            pcall(function() remote:FireServer("buy", petName) end)
            ok = true
        end
    end)
    return ok
end

local function buyAura(auraName)
    if not auraName then
        return false
    end
    local ok = false
    pcall(function()
        local remote = ReplicatedStorage:FindFirstChild("cPetShopRemote")
            or ReplicatedStorage:FindFirstChild("cAuraShopRemote")
            or (RemoteEvents and RemoteEvents:FindFirstChild("buyAuraRemote"))
        if remote and remote:IsA("RemoteFunction") then
            pcall(function() remote:InvokeServer(auraName) end)
            pcall(function() remote:InvokeServer("buy", auraName) end)
            pcall(function() remote:InvokeServer("buyAura", auraName) end)
            ok = true
        elseif remote and remote:IsA("RemoteEvent") then
            pcall(function() remote:FireServer(auraName) end)

local bossRarities = {"Any", "Common", "Rare", "Epic", "Legendary", "Mythic", "Rainbow"}
getgenv().selectedBossRarity = "Any"
getgenv().autoKillBoss = false
getgenv().autoBossChest = true

getgenv().lockedBoss = nil
getgenv().bossHeight = 34

local playerNameSet = {}
local function refreshPlayerNames()
    playerNameSet = {}
    for _, plr in ipairs(Players:GetPlayers()) do
        playerNameSet[plr.Name] = true
        if plr.Character then
            playerNameSet[plr.Character.Name] = true
        end
    end
end

local function getBossRoot(model)
    if not model then
        return nil
    end
    return model:FindFirstChild("HumanoidRootPart")
        or model:FindFirstChild("Torso")
        or model:FindFirstChild("UpperTorso")
        or model:FindFirstChild("Root")
        or model.PrimaryPart
        or model:FindFirstChildWhichIsA("BasePart", true)
end

local function isLivingBoss(model)
    if not model or not model:IsA("Model") then
        return false
    end
    if playerNameSet[model.Name] or model == LocalPlayer.Character then
        return false
    end
    local hum = model:FindFirstChildOfClass("Humanoid")
    if not hum or hum.Health <= 0 then
        return false
    end
    local root = getBossRoot(model)
    if not root then
        return false
    end
    return true, hum, root
end

local function rarityMatches(name)
    local rarity = getgenv().selectedBossRarity or "Any"
    if rarity == "Any" then
        return true
    end
    local n, r = string.lower(name), string.lower(rarity)
    if r == "legendary" or r == "legend" then
        return string.find(n, "legend") ~= nil
    end
    if r == "mythic" or r == "myth" then
        return string.find(n, "myth") ~= nil
    end
    if r == "rainbow" then
        return string.find(n, "rainbow") ~= nil or string.find(n, "admin") ~= nil
    end
    return string.find(n, r) ~= nil
end

local function scoreBossCandidate(model, myPos)
    local ok, hum, root = isLivingBoss(model)
    if not ok then
        return nil
    end
    local n = string.lower(model.Name)
    local parentN = model.Parent and string.lower(model.Parent.Name) or ""
    local looksBoss = string.find(n, "boss")
        or string.find(n, "raid")
        or string.find(n, "king")
        or string.find(n, "lord")
        or string.find(n, "titan")
        or string.find(n, "golem")
        or string.find(parentN, "boss")
        or string.find(parentN, "raid")
        or (hum.MaxHealth >= 1000)
    if not looksBoss then
        return nil
    end
    if getgenv().selectedBossRarity ~= "Any"
        and not rarityMatches(model.Name)
        and not rarityMatches(parentN) then
        return nil
    end
    local dist = (root.Position - myPos).Magnitude
    local score = hum.MaxHealth - dist
    if string.find(n, "boss") or string.find(parentN, "boss") then
        score = score + 50000
    end
    return score, model, hum, root
end

-- Prefer LoYaL targets: BossArena / Boss / BossDamageHitbox
local function findBossHitbox()
    for _, obj in ipairs(workspace:GetDescendants()) do
        if obj.Name == "BossDamageHitbox" and obj:IsA("BasePart") then
            return obj
        end
    end
    return nil
end

local function findBossModel()
    refreshPlayerNames()
    local char = LocalPlayer.Character
    local myRoot = char and char:FindFirstChild("HumanoidRootPart")
    local myPos = myRoot and myRoot.Position or Vector3.new(0, 0, 0)

    if getgenv().lockedBoss and getgenv().lockedBoss.Parent then
        if isLivingBoss(getgenv().lockedBoss) then
            return getgenv().lockedBoss
        end
        getgenv().lockedBoss = nil
    end

    -- priority: exact names used by working hubs
    for _, name in ipairs({"Boss", "BossArena"}) do
        local m = workspace:FindFirstChild(name, true)
        if m and m:IsA("Model") and isLivingBoss(m) then
            if rarityMatches(m.Name) or getgenv().selectedBossRarity == "Any" then
                getgenv().lockedBoss = m
                return m
            end
        end
    end

    local best, bestScore = nil, -math.huge
    local searchRoots = {workspace}
    for _, name in ipairs({"Bosses", "Boss", "BossArena", "Raids", "Raid", "NPCs", "Mobs", "Enemies", "Monsters"}) do
        local f = workspace:FindFirstChild(name)
        if f then
            table.insert(searchRoots, f)
        end
    end

    for _, rootFolder in ipairs(searchRoots) do
        for _, obj in ipairs(rootFolder:GetDescendants()) do
            if obj:IsA("Model") then
                local score, model = scoreBossCandidate(obj, myPos)
                if score and score > bestScore then
                    bestScore = score
                    best = model
                end
            end
        end
    end

    if best then
        getgenv().lockedBoss = best
    end
    return best
end

local function bossHeavy()
    local ev = LocalPlayer:FindFirstChild("muscleEvent") or MuscleEvent
    if not ev then
        return
    end
    pcall(function()
        local bag = LocalPlayer:FindFirstChild("Backpack")
        local char = LocalPlayer.Character

        end

        ev:FireServer("slam")
        equipPunch()
    end)
end

local function ensureBossAntiKnock(myRoot)
    if not myRoot then
        return
    end
    local bv = myRoot:FindFirstChild("DAN_BOSS_BV")
    if not bv then
        bv = Instance.new("BodyVelocity")
        bv.Name = "DAN_BOSS_BV"
        bv.MaxForce = Vector3.new(1e5, 1e5, 1e5)
        bv.P = 1250
        bv.Parent = myRoot
    end
    bv.Velocity = Vector3.new(0, 0, 0)
end

local function clearBossAntiKnock(myRoot)
    if myRoot then
        local bv = myRoot:FindFirstChild("DAN_BOSS_BV")
        if bv then
            bv:Destroy()
        end
    end
end

-- Old-style: in FRONT of boss + higher (not straight up into water/void)
local function bossStick(myRoot, bossRoot)
    local h = tonumber(getgenv().bossHeight) or 10
    local range = tonumber(getgenv().bossRange) or 8
    local look = bossRoot.CFrame.LookVector
    if look.Magnitude < 0.05 then
        look = Vector3.new(0, 0, 1)
    else
        look = look.Unit
    end
    -- stand in front of boss, elevated
    local pos = bossRoot.Position - look * range + Vector3.new(0, h, 0)
    myRoot.CFrame = CFrame.new(pos, bossRoot.Position + Vector3.new(0, h * 0.3, 0))
    pcall(function()
        myRoot.AssemblyLinearVelocity = Vector3.new(0, 0, 0)
        myRoot.Velocity = Vector3.new(0, 0, 0)
    end)
    ensureBossAntiKnock(myRoot)
end

local function bossDodge(myRoot, bossRoot)
    local h = tonumber(getgenv().bossHeight) or 10
    local range = tonumber(getgenv().bossRange) or 8
    local side = Vector3.new(1, 0, 0)
    local away = (myRoot.Position - bossRoot.Position)
    if away.Magnitude > 0.1 then
        side = away:Cross(Vector3.new(0, 1, 0))
        if side.Magnitude > 0.1 then
            side = side.Unit
        end
    end
    local dir = (math.random(0, 1) == 0) and 1 or -1
    local pos = bossRoot.Position + side * (range + 6) * dir + Vector3.new(0, h, 0)
    myRoot.CFrame = CFrame.new(pos, bossRoot.Position)
    ensureBossAntiKnock(myRoot)
end

-- Huge hitbox reach: touch all boss parts + named hitbox
local function hitBossFromRange(myRoot, bossModel, hitbox)
    if not myRoot then
        return
    end
    local targets = {}
    if hitbox then
        table.insert(targets, hitbox)
    end
    if bossModel then
        for _, p in ipairs(bossModel:GetDescendants()) do
            if p:IsA("BasePart") then
                table.insert(targets, p)
            end
        end
        local root = getBossRoot(bossModel)
        if root then
            table.insert(targets, root)
        end
    end
    for _, obj in ipairs(workspace:GetDescendants()) do
        local n = obj.Name
        if (n == "BossDamageHitbox" or n == "BossChest" or n == "Boss") and obj:IsA("BasePart") then
            table.insert(targets, obj)
        end
    end
    if firetouchinterest then
        for _, part in ipairs(targets) do
            pcall(function()
                firetouchinterest(part, myRoot, 0)
                firetouchinterest(part, myRoot, 1)
                firetouchinterest(part, myRoot, 0)
                firetouchinterest(part, myRoot, 1)
            end)
        end
    end
end

-- Dive close to land hits, then return high (fixes no damage at high height)
local function bossDiveHit(myRoot, targetPart, boss, hitbox)
    if not myRoot or not targetPart then
        return
    end
    local highCF = myRoot.CFrame
    -- snap onto boss for damage window
    myRoot.CFrame = CFrame.new(targetPart.Position + Vector3.new(0, 3, 0), targetPart.Position)
    pcall(function()
        myRoot.AssemblyLinearVelocity = Vector3.new(0, 0, 0)
    end)
    hitBossFromRange(myRoot, boss, hitbox)
    bossPunchBurst()
    hitBossFromRange(myRoot, boss, hitbox)
    -- return to safe height
    bossStick(myRoot, targetPart)
end

-- LoYaL chest: BossChest + bossChestPrompt
local function claimBossChest()
    if not getgenv().autoBossChest then
        return
    end
    local myRoot = LocalPlayer.Character and LocalPlayer.Character:FindFirstChild("HumanoidRootPart")
    if not myRoot then
        return
    end

    -- exact names first
    local exactNames = {"BossChest", "bossChestPrompt"}
    for _, name in ipairs(exactNames) do
        for _, obj in ipairs(workspace:GetDescendants()) do
            if obj.Name == name then
                local part = obj:IsA("BasePart") and obj or obj:FindFirstChildWhichIsA("BasePart", true)
                local prompt = obj:IsA("ProximityPrompt") and obj or obj:FindFirstChildWhichIsA("ProximityPrompt", true)
                if part then
                    pcall(function()
                        myRoot.CFrame = part.CFrame + Vector3.new(0, 3, 0)
                    end)
                end
                if prompt and fireproximityprompt then
                    pcall(function()
                        fireproximityprompt(prompt)
                    end)
                end
                if part and firetouchinterest then
                    pcall(function()
                        firetouchinterest(part, myRoot, 0)
                        task.wait(0.03)
                        firetouchinterest(part, myRoot, 1)
                    end)
                end
            end
        end
    end

    -- fallback scan
    for _, obj in ipairs(workspace:GetDescendants()) do
        local n = string.lower(obj.Name)
        if string.find(n, "bosschest") or (string.find(n, "chest") and string.find(n, "boss"))
            or n == "bosschestprompt" then
            local part = obj:IsA("BasePart") and obj or obj:FindFirstChildWhichIsA("BasePart", true)
            local prompt = obj:IsA("ProximityPrompt") and obj or obj:FindFirstChildWhichIsA("ProximityPrompt", true)
            if prompt and fireproximityprompt then
                pcall(function()
                    if part then
                        myRoot.CFrame = part.CFrame + Vector3.new(0, 3, 0)
                    end
                    fireproximityprompt(prompt)
                end)
            elseif part and firetouchinterest then
                pcall(function()
                    myRoot.CFrame = part.CFrame + Vector3.new(0, 3, 0)
                    firetouchinterest(part, myRoot, 0)
                    task.wait(0.03)
                    firetouchinterest(part, myRoot, 1)
                end)
            end
        end
        if obj:IsA("ProximityPrompt") then
            local pn = string.lower(obj.Name)
            local parentN = string.lower(obj.Parent and obj.Parent.Name or "")
            if string.find(pn, "boss") or string.find(parentN, "boss") or string.find(parentN, "chest") then
                if fireproximityprompt then
                    pcall(function()
                        fireproximityprompt(obj)
                    end)
                end
            end
        end
    end
end

local function readLocalStats()
    local stats = {
        Strength = 0,
        Durability = 0,
        Rebirths = 0,
        Kills = 0,
        Brawls = 0
    }
    pcall(function()
        local ls = LocalPlayer:FindFirstChild("leaderstats")
        if ls then
            for _, v in ipairs(ls:GetChildren()) do
                local n = string.lower(v.Name)
                local val = tonumber(v.Value) or 0
                if string.find(n, "stren") then
                    stats.Strength = val
                elseif string.find(n, "dura") then
                    stats.Durability = val
                elseif string.find(n, "rebirth") then
                    stats.Rebirths = val
                elseif string.find(n, "kill") then
                    stats.Kills = val
                elseif string.find(n, "brawl") then
                    stats.Brawls = val
                end
            end
        end
        -- Muscle Legends often stores more under player folders
        for _, folderName in ipairs({"MuscleStats", "stats", "Stats", "playerStats"}) do
            local f = LocalPlayer:FindFirstChild(folderName)
            if f then
                for _, v in ipairs(f:GetDescendants()) do
                    if v:IsA("NumberValue") or v:IsA("IntValue") then
                        local n = string.lower(v.Name)
                        local val = tonumber(v.Value) or 0
                        if string.find(n, "stren") and val > stats.Strength then
                            stats.Strength = val
                        elseif string.find(n, "dura") and val > stats.Durability then
                            stats.Durability = val
                        elseif string.find(n, "rebirth") and val > stats.Rebirths then
                            stats.Rebirths = val
                        end
                    end
                end
            end
        end
    end)
    return stats
end

local function formatNum(n)
    n = tonumber(n) or 0
    local abs = math.abs(n)
    if abs >= 1e12 then
        return string.format("%.2fT", n / 1e12)
    elseif abs >= 1e9 then
        return string.format("%.2fB", n / 1e9)
    elseif abs >= 1e6 then
        return string.format("%.2fM", n / 1e6)
    elseif abs >= 1e3 then
        return string.format("%.2fK", n / 1e3)
    end
    return string.format("%.0f", n)
end

local statStart = readLocalStats()
local statStartTime = os.clock()
local lastStatSnapshot = {
    Strength = statStart.Strength,
    Durability = statStart.Durability
}

local function ratesFor(delta, seconds)
    if seconds <= 0 then
        return 0, 0, 0, 0, 0, 0
    end
    local perSec = delta / seconds
    return perSec * 60, perSec * 3600, perSec * 86400, perSec * 604800, perSec * 2592000, perSec * 31536000
end

-- STAT SPY (other players)

local function readPlayerStats(player)
    local stats = {
        Strength = nil,
        Durability = nil,
        Agility = nil,
        Rebirths = nil,
        Kills = nil,
        Brawls = nil
    }
    if not player then
        return stats
    end
    pcall(function()
        local function applyName(n, val)
            local low = string.lower(tostring(n))
            if string.find(low, "stren") then
                stats.Strength = val
            elseif string.find(low, "dura") then
                stats.Durability = val
            elseif string.find(low, "agil") or string.find(low, "speed") then
                stats.Agility = val
            elseif string.find(low, "rebirth") then
                stats.Rebirths = val
            elseif string.find(low, "kill") then
                stats.Kills = val
            elseif string.find(low, "brawl") then
                stats.Brawls = val
            else
                stats[tostring(n)] = val
            end
        end
        local ls = player:FindFirstChild("leaderstats")
        if ls then
            for _, v in ipairs(ls:GetChildren()) do
                if v:IsA("NumberValue") or v:IsA("IntValue") or v:IsA("StringValue") then
                    applyName(v.Name, v.Value)
                end
            end
        end
        for _, folderName in ipairs({"MuscleStats", "stats", "Stats", "playerStats", "Data", "data"}) do
            local f = player:FindFirstChild(folderName)
            if f then
                for _, v in ipairs(f:GetDescendants()) do
                    if v:IsA("NumberValue") or v:IsA("IntValue") then
                        applyName(v.Name, v.Value)
                    end
                end
            end
        end
        -- common alternate attribute names
        for _, attr in ipairs({"Durability", "durability", "Dura", "dura", "Strength", "strength", "Agility", "agility"}) do
            local ok, val = pcall(function()
                return player:GetAttribute(attr)
            end)
            if ok and val ~= nil then
                applyName(attr, val)
            end
        end
    end)
    return stats
end

local function playerList()
    local names = {}
    for _, p in ipairs(Players:GetPlayers()) do
        if p ~= LocalPlayer then
            names[#names + 1] = p.Name
        end
    end
    table.sort(names)
    if #names == 0 then
        names[1] = "(no other players)"
    end
    return names
end

                lines[#lines + 1] = tostring(k) .. ": " .. formatNum(v)
                count = count + 1
            end
        end
        if count == 0 then
            lines[#lines + 1] = "No public stats found (may be hidden server-side)"
        end
        if stats.Durability == nil then
            lines[#lines + 1] = "Durability: (not visible on this player)"
        end
        pcall(function()
            spyResult:SetDesc(table.concat(lines, "\n"))
        end)
        WindUI:Notify({
