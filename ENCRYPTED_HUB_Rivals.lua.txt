local function notify(t, x)
	pcall(function() StarterGui:SetCore("SendNotification", {Title=tostring(t),Text=tostring(x),Duration=3}) end)
end
local function getChar() return LP.Character end
local function getHRP() local c=getChar() return c and c:FindFirstChild("HumanoidRootPart") end
local function getHum() local c=getChar() return c and c:FindFirstChildOfClass("Humanoid") end
local function getRoot(plr) local c=plr and plr.Character return c and c:FindFirstChild("HumanoidRootPart") end

local speedC
local function setSpeed(on)
	S.speedOn = on
	if speedC then pcall(function() speedC:Disconnect() end) speedC = nil end
	if on then
		speedC = RunService.Heartbeat:Connect(function()
			if not S.speedOn then return end
			local h = getHum()
			if h then h.WalkSpeed = S.speed end
		end)
	end
end
local function setJump(on)
	S.jumpOn = on
	if on then
		local h = getHum()
		if h then h.JumpPower = S.jump end
	end
end
local flyBV, flyBG
local function setFly(on)
	S.fly = on
	local hrp = getHRP()
	if flyBV then pcall(function() flyBV:Destroy() end) flyBV=nil end
	if flyBG then pcall(function() flyBG:Destroy() end) flyBG=nil end
	if on and hrp then
		flyBV = Instance.new("BodyVelocity")
		flyBV.MaxForce = Vector3.new(1e5,1e5,1e5)
		flyBV.Velocity = Vector3.zero
		flyBV.Parent = hrp
		flyBG = Instance.new("BodyGyro")
		flyBG.MaxTorque = Vector3.new(1e5,1e5,1e5)
		flyBG.P = 3000
		flyBG.Parent = hrp
	end
end
local noclipC
local function setNoclip(on)
	S.noclip = on
	if noclipC then pcall(function() noclipC:Disconnect() end) noclipC=nil end
	if on then
		noclipC = RunService.Stepped:Connect(function()
			if not S.noclip then return end
			local c = getChar()
			if not c then return end
			for _, p in ipairs(c:GetDescendants()) do
				if p:IsA("BasePart") then p.CanCollide = false end
			end
		end)
	end
end
local espFolder
local function clearESP()
	if espFolder then pcall(function() espFolder:Destroy() end) end
	espFolder = nil
end
local function setESP(on)
	S.esp = on
	clearESP()
	if not on then return end
	espFolder = Instance.new("Folder")
	espFolder.Name = "EH_ESP"
	espFolder.Parent = CoreGui
	local function bill(plr)
		if plr == LP or not plr.Character then return end
		local hrp = plr.Character:FindFirstChild("HumanoidRootPart")
		if not hrp then return end
		local bg = Instance.new("BillboardGui")
		bg.Size = UDim2.fromOffset(100, 30)
		bg.AlwaysOnTop = true
		bg.Adornee = hrp
		bg.Parent = espFolder
		local t = Instance.new("TextLabel", bg)
		t.Size = UDim2.fromScale(1,1)
		t.BackgroundTransparency = 1
		t.Text = plr.Name
		t.TextColor3 = Color3.fromRGB(100, 180, 255)
		t.TextStrokeTransparency = 0.5
		t.Font = Enum.Font.GothamBold
		t.TextSize = 12
	end
	for _, p in ipairs(Players:GetPlayers()) do bill(p) end
end

UserInputService.JumpRequest:Connect(function()
	if S.infJump then
		local h = getHum()
		if h then h:ChangeState(Enum.HumanoidStateType.Jumping) end
	end
end)
UserInputService.InputBegan:Connect(function(input, gp)
	if gp then return end
	if S.clickTp and input.UserInputType == Enum.UserInputType.MouseButton1 then
		local mouse = LP:GetMouse()
		local hrp = getHRP()
		if mouse.Hit and hrp then hrp.CFrame = CFrame.new(mouse.Hit.Position + Vector3.new(0, 3, 0)) end
	end
end)

RunService.RenderStepped:Connect(function()
	if not S.alive then return end
	if S.fly and flyBV then
		local cam = Workspace.CurrentCamera
		local dir = Vector3.zero
		if UserInputService:IsKeyDown(Enum.KeyCode.W) then dir = dir + cam.CFrame.LookVector end
		if UserInputService:IsKeyDown(Enum.KeyCode.S) then dir = dir - cam.CFrame.LookVector end
		if UserInputService:IsKeyDown(Enum.KeyCode.A) then dir = dir - cam.CFrame.RightVector end
		if UserInputService:IsKeyDown(Enum.KeyCode.D) then dir = dir + cam.CFrame.RightVector end
		if UserInputService:IsKeyDown(Enum.KeyCode.Space) then dir = dir + Vector3.new(0,1,0) end
		if UserInputService:IsKeyDown(Enum.KeyCode.LeftControl) then dir = dir - Vector3.new(0,1,0) end
		if dir.Magnitude > 0 then dir = dir.Unit * S.flySpeed end
		flyBV.Velocity = dir
		if flyBG then flyBG.CFrame = cam.CFrame end
	end
	if S.fullbright then
		pcall(function() Lighting.Brightness = 2 Lighting.ClockTime = 14 Lighting.FogEnd = 1e6 end)
	end
	if S.noFog then pcall(function() Lighting.FogEnd = 1e6 end) end
end)

local function numberBox(page, label, order, default, minV, maxV, onApply)
	local hold = Instance.new("Frame")
	hold.Size = UDim2.new(1,-8,0,56)
	hold.BackgroundColor3 = Color3.fromRGB(18,20,36)
	hold.LayoutOrder = order
	hold.Parent = page
	corner(hold,8)
	local lab = Instance.new("TextLabel", hold)
	lab.Size = UDim2.new(1,-16,0,18); lab.Position = UDim2.fromOffset(8,4)
	lab.BackgroundTransparency = 1; lab.Text = label
	lab.Font = Enum.Font.Gotham; lab.TextSize = 11; lab.TextColor3 = COLORS.muted
	lab.TextXAlignment = Enum.TextXAlignment.Left
	local box = Instance.new("TextBox", hold)
	box.Size = UDim2.new(1,-90,0,26); box.Position = UDim2.fromOffset(8,24)
	box.BackgroundColor3 = COLORS.panel; box.Text = tostring(default)
	box.Font = Enum.Font.GothamMedium; box.TextSize = 13; box.TextColor3 = COLORS.text
	box.ClearTextOnFocus = false; corner(box,6)
	local apply = Instance.new("TextButton", hold)
	apply.Size = UDim2.fromOffset(70,26); apply.Position = UDim2.new(1,-78,0,24)
	apply.BackgroundColor3 = COLORS.accent; apply.Text = "Apply"
	apply.Font = Enum.Font.GothamBold; apply.TextSize = 12; apply.TextColor3 = Color3.new(1,1,1)
	corner(apply,6)
	local function doApply()
		local n = tonumber(box.Text)
		if not n then notify("Input","Number") return end
		n = math.clamp(n, minV, maxV)
		box.Text = tostring(n)
		if onApply then pcall(onApply, n) end
	end
	apply.MouseButton1Click:Connect(doApply)
	box.FocusLost:Connect(function(e) if e then doApply() end end)
end

		end
	end

end
