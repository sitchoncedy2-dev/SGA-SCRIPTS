local function findHRP()
	local c = LP.Character
	return c and c:FindFirstChild("HumanoidRootPart")
end

	local c = LP.Character
	return c and c:FindFirstChildOfClass("Humanoid")

local function GetNet(name)
	local ok, res = pcall(function()
		local net = RS:FindFirstChild("Packages") and RS.Packages:FindFirstChild("Networking")
		if net and net:FindFirstChild(name) then return net:FindFirstChild(name) end
		local r = RS:FindFirstChild(name, true)
		return r
	end)
	return ok and res or nil
end

local function firePrompt(prompt)
	if not prompt then return end
	pcall(function()
		prompt.HoldDuration = 0
		prompt.MaxActivationDistance = 999
		if fireproximityprompt then
			fireproximityprompt(prompt)
		end
	end)
end

-- Find stealable brainrot prompts on plots / podiums
local function FindStealPrompts()
	local found = {}
	local function scan(root)
		if not root then return end
		for _, d in ipairs(root:GetDescendants()) do
			if d:IsA("ProximityPrompt") then
				local act = string.lower(d.ActionText or "")
				local obj = string.lower(d.ObjectText or "")
				local nm = string.lower(d.Name or "")
				if act:find("steal") or act:find("take") or act:find("grab")
					or obj:find("brainrot") or nm:find("steal")
					or (d.Enabled and (act:find("pick") or act:find("collect"))) then
					local part = d.Parent
					if part:IsA("Attachment") then part = part.Parent end
					if part and part:IsA("BasePart") then
						found[#found + 1] = { prompt = d, part = part, label = d.ObjectText or d.ActionText or "Brainrot" }
					end
				end
			end
		end
	end

	pcall(function() scan(Workspace:FindFirstChild("Plots")) end)
	pcall(function() scan(Workspace:FindFirstChild("Bases")) end)
	pcall(function() scan(Workspace:FindFirstChild("AnimalPodiums")) end)
	pcall(function()
		for _, m in ipairs(Workspace:GetChildren()) do
			if m:IsA("Model") or m:IsA("Folder") then
				local n = string.lower(m.Name)
				if n:find("plot") or n:find("base") or n:find("podium") or n:find("animal") then
					scan(m)
				end
			end
		end
	end)

	-- fallback: any enabled prompt near other bases
	if #found == 0 then
		pcall(function()
			for _, d in ipairs(Workspace:GetDescendants()) do
				if d:IsA("ProximityPrompt") and d.Enabled then
					local part = d.Parent
					if part:IsA("Attachment") then part = part.Parent end
					if part and part:IsA("BasePart") then
						found[#found + 1] = { prompt = d, part = part, label = d.ObjectText or d.ActionText or "?" }
					end
				end
			end
		end)
	end
	return found
end

local function IsOwnBase(part)
	if not part then return false end
	local hrp = findHRP()
	-- attribute owner checks
	local model = part
	for _ = 1, 6 do
		if not model then break end
		local owner = model:GetAttribute("Owner") or model:GetAttribute("OwnerName") or model:GetAttribute("Player")
		if owner and (owner == LP.Name or owner == LP.UserId) then return true end
		model = model.Parent
	end
	return false
end

local function GetPriorityPrompt(list)
	if #priorityTargets > 0 then
		for _, item in ipairs(list) do
			local model = item.part
			for _ = 1, 8 do
				if not model then break end
				local owner = model:GetAttribute("Owner") or model:GetAttribute("OwnerName")
				if owner then
					for _, name in ipairs(priorityTargets) do
						if tostring(owner) == name or string.find(string.lower(tostring(owner)), string.lower(name), 1, true) then
							return item
						end
					end
				end
				-- check nearby player name in model name
				local mn = string.lower(model.Name)
				for _, name in ipairs(priorityTargets) do
					if mn:find(string.lower(name), 1, true) then
						return item
					end
				end
				model = model.Parent
			end
		end
	end
	-- non-own first
	for _, item in ipairs(list) do
		if not IsOwnBase(item.part) then
			return item
		end
	end
	return list[1]
end

local function TweenTo(pos)
	local hrp = findHRP()
	if not hrp or not pos then return end
	local dist = (hrp.Position - pos).Magnitude
	local t = math.clamp(dist / math.max(tpSpeed, 50), 0.05, 2.5)
	local tw = TweenService:Create(hrp, TweenInfo.new(t, Enum.EasingStyle.Linear), {
		CFrame = CFrame.new(pos + Vector3.new(0, 2, 0)),
	})
	tw:Play()
	tw.Completed:Wait()
	hrp.AssemblyLinearVelocity = Vector3.zero
end

local function InstantTP(pos)
	local hrp = findHRP()
	if not hrp or not pos then return end
	hrp.CFrame = CFrame.new(pos + Vector3.new(0, 2, 0))
	hrp.AssemblyLinearVelocity = Vector3.zero
end

local function DoSteal(item, instant)
	if not item or not item.prompt then return false end
	local pos = item.part.Position
	if instant then
		InstantTP(pos)
	else
		TweenTo(pos)
	end
	task.wait(0.12)
	firePrompt(item.prompt)
	task.wait(0.15)
	firePrompt(item.prompt)
	-- also try common steal remotes
	pcall(function()
		for _, name in ipairs({
			"Steal", "StealAnimal", "TakeBrainrot", "ClaimBrainrot",
			"RE/Steal", "RF/Steal",
		}) do
			local r = GetNet(name)
			if r then
				if r:IsA("RemoteEvent") then
					r:FireServer()
				elseif r:IsA("RemoteFunction") then
					r:InvokeServer()
				end
			end
		end
	end)
	return true
end

local function InstantSteal()
	local list = FindStealPrompts()
	if #list == 0 then
		notify("Steal", "No brainrots found")
		return false
	end
	local target = GetPriorityPrompt(list)
	if not target then
		notify("Steal", "No valid target")
		return false
	end
	notify("Steal", tostring(target.label))
	return DoSteal(target, true)
end

local function StealSlot(slot)
	local list = FindStealPrompts()
	if #list == 0 then return InstantSteal() end
	-- try index as slot
	local idx = math.clamp(tonumber(slot) or 1, 1, #list)
	return DoSteal(list[idx], true)
end

local function SelfRagdoll()
	local h = findHum()
	if h then pcall(function() h:ChangeState(Enum.HumanoidStateType.Physics) end) end
end

local function InstantReset()
	local h = findHum()
	if h then pcall(function() h.Health = 0 end) end
end

local function RagdollTech()
	SelfRagdoll()
	task.wait(0.12)
	local h = findHum()
	if h then pcall(function() h:ChangeState(Enum.HumanoidStateType.GettingUp) end) end
end

local function Rejoin()
	pcall(function() TeleportService:Teleport(game.PlaceId, LP) end)
end

local function Leave()
	pcall(function() game:Shutdown() end)
end

local function GetClosestPlayer()
	local hrp = findHRP()
	if not hrp then return nil end
	local best, bestD = nil, math.huge
	for _, p in ipairs(Players:GetPlayers()) do
		if p ~= LP and p.Character then
			local t = p.Character:FindFirstChild("HumanoidRootPart")
			if t then
				local d = (t.Position - hrp.Position).Magnitude
				if d < bestD then bestD, best = d, p end
			end
		end
	end
	return best
end

local function SpamClosest()
	local t = GetClosestPlayer()
	if not t then
		notify("Spam", "No player")
		return
	end
	notify("Spam", t.Name .. " | " .. spamCmd)
	local thrp = t.Character and t.Character:FindFirstChild("HumanoidRootPart")
	if thrp then InstantTP(thrp.Position) end
	for _ = 1, 10 do
		if spamCmd == "ragdoll" then
			SelfRagdoll()
		elseif spamCmd == "steal" then
			InstantSteal()
		else
			-- bat / hit style if remote exists
			pcall(function()
				local r = GetNet("RE/BatSwing/Trigger") or GetNet("Hit") or GetNet("Attack")
				if r and r:IsA("RemoteEvent") then r:FireServer() end
			end)
		end
		task.wait(0.1)
	end
end

local function TPSlot(n)
	local plots = Workspace:FindFirstChild("Plots") or Workspace:FindFirstChild("Bases")
	if plots then
		local p = plots:FindFirstChild(tostring(n)) or plots:FindFirstChild("Plot" .. tostring(n))
		if p then
			local pos = p:GetPivot().Position
			InstantTP(pos)
			notify("Slot", "TP plot " .. tostring(n))
			return
		end
	end
	-- fallback: prompt list index
	local list = FindStealPrompts()
	if list[n] then
		InstantTP(list[n].part.Position)
		notify("Slot", "TP target " .. tostring(n))
	else
		notify("Slot", "Not found")
	end
end

-- Loops
task.spawn(function()
	while not dead do
		if autoSteal or semiInstant then
			pcall(InstantSteal)
			task.wait(stealDelay)
		else
			task.wait(0.3)
		end
	end
end)

task.spawn(function()
	while not dead do
		task.wait(0.35)
		if speedEnabled then
			local h = findHum()
			if h then pcall(function() h.WalkSpeed = speedValue end) end
		end
	end
end)

task.spawn(function()
	while not dead do
		task.wait(0.05)
		if (antiRagdoll or autoDefense) then
			local h = findHum()
			if h and h:GetState() == Enum.HumanoidStateType.Physics then
				pcall(function() h:ChangeState(Enum.HumanoidStateType.GettingUp) end)
			end
		end
	end
end)

task.spawn(function()
	while not dead do
		if spamClosest then
			pcall(SpamClosest)
			task.wait(2)
		else
			task.wait(0.4)
		end
	end
end)

pcall(function()
	track(UIS.JumpRequest:Connect(function()
		if not infJump then return end
		local h = findHum()
		if h then pcall(function() h:ChangeState(Enum.HumanoidStateType.Jumping) end) end
	end))
end)
