-- ================= FEATURES (loaded only AFTER unlock) =================
local function bootFeatures()
	if unlocked then return end
	unlocked = true

	local F = {
		alive = true,
		autoFarm = false,
		autoSteal = false,
		autoTreadmill = false,
		speedOn = false, speedVal = 18,
		espOn = false,
		minRarity = "Rare",
		stealDelay = 0.7,
		status = "Idle",
		safeMode = true,
		maxStealsPerMinute = 5,
		_stealTimes = {},
	}

	local function safeWait(a, b)
		a, b = a or 0.3, b or 0.8
		if b < a then a, b = b, a end
		task.wait(a + math.random() * (b - a))
	end

	local function canStealNow()
		local now = os.clock()
		local kept = {}
		for _, t0 in ipairs(F._stealTimes) do
			if now - t0 < 60 then table.insert(kept, t0) end
		end
		F._stealTimes = kept
		if #kept >= F.maxStealsPerMinute then return false end
		table.insert(F._stealTimes, now)
		return true
	end

	local function getHRP()
		local c = LP.Character
		return c and c:FindFirstChild("HumanoidRootPart")
	end
	local function getHum()
		local c = LP.Character
		return c and c:FindFirstChildOfClass("Humanoid")
	end

	local function firePrompt(prompt)
		if not prompt then return end
		pcall(function()
			-- never permanently change HoldDuration
			if fireproximityprompt then
				fireproximityprompt(prompt)
			else
				prompt:InputHoldBegin()
				safeWait(0.08, 0.15)
				prompt:InputHoldEnd()
			end
		end)
	end

	local RarityRank = {
		Common=1, Uncommon=2, Rare=3, Epic=4, Legendary=5, Mythical=6, Mythic=6, Secret=7, Exclusive=8,
	}
	local function rarityScore(name)
		if not name then return 0 end
		name = tostring(name)
		for k,v in pairs(RarityRank) do
			if name:lower():find(k:lower()) then return v end
		end
		return 0
	end

	local function collectEggs()
		local list = {}
		-- only scan prompts (lighter than full EggState require on some ACs)
		for _, d in ipairs(Workspace:GetDescendants()) do
			if d:IsA("ProximityPrompt") and d.Enabled then
				local n = d.Name:lower()
				if n:find("carry") or n:find("egg") or n:find("steal") or n:find("pick") then
					local model = d:FindFirstAncestorOfClass("Model")
					local part = d.Parent
					if part and part:IsA("BasePart") then
						local rname = (model and (model:GetAttribute("Rarity") or model.Name)) or part.Name
						table.insert(list, {
							model = model,
							part = part,
							prompt = d,
							rarity = tostring(rname),
							score = rarityScore(rname) * 100000 + part.Size.Magnitude,
						})
					end
				end
			end
		end
		table.sort(list, function(a,b) return (a.score or 0) > (b.score or 0) end)
		return list
	end

	local function isCarrying()
		local char = LP.Character
		if not char then return false end
		for _, t in ipairs(char:GetChildren()) do
			if t:IsA("Tool") and t.Name:lower():find("egg") then return true end
		end
		return false
	end

	local function stealEgg(entry)
		local hrp = getHRP()
		if not hrp or not entry or not entry.part then return false end
		if not canStealNow() then
			F.status = "Cooldown (safe)"
			safeWait(3, 6)
			return false
		end
		local dist = (hrp.Position - entry.part.Position).Magnitude
		-- if far, walk-ish by single TP only when needed, with long pause
		if dist > 18 then
			local off = Vector3.new(math.random(-12,12)/10, 2.8, math.random(-12,12)/10)
			pcall(function() hrp.CFrame = entry.part.CFrame + off end)
			safeWait(0.4, 0.9)
		end
		firePrompt(entry.prompt)
		safeWait(F.stealDelay, F.stealDelay + 0.5)
		return true
	end

	local function findBase()
		for _, name in ipairs({"Plots", "PlayerPlots", "Bases"}) do
			local f = Workspace:FindFirstChild(name)
			if f then
				local mine = f:FindFirstChild(LP.Name) or f:FindFirstChild(tostring(LP.UserId))
				if mine then
					local p = mine:FindFirstChildWhichIsA("BasePart", true)
					if p then return p end
				end
			end
		end
		return nil
	end

	local function findTreadmill()
		for _, d in ipairs(Workspace:GetDescendants()) do
			if d:IsA("BasePart") and d.Name:lower():find("tread") then return d end
		end
		return nil
	end

	local function goTo(part)
		local hrp = getHRP()
		if not hrp or not part then return end
		pcall(function()
			hrp.CFrame = part.CFrame + Vector3.new(math.random(-8,8)/10, 3, math.random(-8,8)/10)
		end)
		safeWait(0.35, 0.8)
	end

	local function farmTreadmill(seconds)
		local tm = findTreadmill()
		if tm then goTo(tm) end
		local t0 = os.clock()
		local nextFix = 0
		while F.alive and (F.autoFarm or F.autoTreadmill) and os.clock() - t0 < seconds do
			local hum = getHum()
			if hum then
				-- barely above default
				pcall(function()
					if hum.WalkSpeed < 16 then hum.WalkSpeed = 16 end
					if F.speedOn and hum.WalkSpeed < F.speedVal then
						hum.WalkSpeed = math.min(F.speedVal, 20)
					end
					hum:Move(Vector3.new(0.8, 0, 0), true)
				end)
			end
			local hrp = getHRP()
			if hrp and tm and os.clock() > nextFix then
				if (hrp.Position - tm.Position).Magnitude > 35 then
					goTo(tm)
					nextFix = os.clock() + math.random(6, 12)
				end
			end
			safeWait(0.4, 0.9)
		end
	end

	local function stealBest()
		F.status = "Looking for eggs..."
		local eggs = collectEggs()
		local need = RarityRank[F.minRarity] or 3
		for _, e in ipairs(eggs) do
			if rarityScore(e.rarity) >= need then
				F.status = "Trying " .. tostring(e.rarity)
				if stealEgg(e) then
					safeWait(0.5, 1.2)
					if isCarrying() then
						F.status = "Back to base"
						local base = findBase()
						if base then goTo(base) end
						safeWait(0.6, 1.2)
						return true
					end
				end
				safeWait(0.5, 1.0)
			end
		end
		if #eggs > 0 then
			stealEgg(eggs[1])
			safeWait(0.6, 1.2)
		end
		return false
	end

	local function runAutoFarm()
		while F.alive and F.autoFarm do
			F.status = "Steal phase"
			stealBest()
			if not F.autoFarm then break end
			safeWait(1.5, 3)
			F.status = "Treadmill phase"
			farmTreadmill(35 + math.random(10, 25))
			if not F.autoFarm then break end
			F.status = "Resting"
			safeWait(2, 5)
		end
		F.status = "Idle"
	end

	end)
