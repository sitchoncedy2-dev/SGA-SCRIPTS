-- Egg value estimate (Chilli scale^1.85 style)
local function eggValue(d)
	if type(d) ~= "table" then return 0 end
	local rate = tonumber(d.EarningRate or d.earningRate or d.value or d.Value) or 0
	local scale = tonumber(d.Scale or d.scale or d.Size or 1) or 1
	local factor
	if scale > 5 then
		factor = (scale / 5) ^ 1.2 * 19.6378
	else
		factor = scale ^ 1.85
	end
	local mut = 1
	if Mutations and type(Mutations.EarningsFor) == "function" then
		local ok, m = pcall(function() return Mutations.EarningsFor(d.Mutation or d.mutation) end)
		if ok and type(m) == "number" then mut = m end
	end
	return rate * factor * mut
end

-- ReadFieldEggs + snapshot fallback
local function collectEggs()
	local list = {}
	local data
	if EggState and type(EggState.ReadFieldEggs) == "function" then
		local s, d = pcall(function() return EggState.ReadFieldEggs() end)
		if s then data = d end
	end
	if type(data) ~= "table" then
		pcall(function() invoke("RF/EggWorld/AskFieldEggSnapshot") end)
		task.wait(0.15)
		if EggState and type(EggState.ReadFieldEggs) == "function" then
			local s, d = pcall(function() return EggState.ReadFieldEggs() end)
			if s then data = d end
		end
	end
	if type(data) == "table" then
		for uid, d in pairs(data) do
			if type(d) == "table" then
				local state = tostring(d.State or d.state or "")
				if state == "Claimed" then continue end
				local model = d.model or d.Model
				local rarity = d.rarity or d.Rarity or (d.record and (d.record.Rarity or d.record.Category)) or "Common"
				local part = model and model:FindFirstChildWhichIsA("BasePart", true)
				if not part and d.BottomCFrame then
					-- synthetic position
				end
				local val = eggValue(d)
				if part then
					local cd = eggCooldown[tostring(uid)]
					if not cd or (os.clock() - cd) > 20 then
						table.insert(list, {
							uid = tostring(uid), model = model, part = part,
							rarity = tostring(rarity), value = val,
							score = rarityScore(rarity) * 1e9 + val,
						})
					end
				end
			end
		end
	end
	-- proximity fallback
	for _, d in ipairs(Workspace:GetDescendants()) do
		if d:IsA("ProximityPrompt") and d.Enabled then
			local n = d.Name
			if n == "CarryAreaEgg" or n:lower():find("carry") then
				local model = d:FindFirstAncestorOfClass("Model")
				local part = d.Parent
				if part and part:IsA("BasePart") then
					local already = false
					for _, e in ipairs(list) do if e.part == part then already = true break end end
					if not already then
						local rname = (model and (model:GetAttribute("Rarity") or model.Name)) or part.Name
						table.insert(list, {
							model = model, part = part, prompt = d,
							rarity = tostring(rname), value = 0,
							score = rarityScore(rname) * 1e9,
						})
					end
				end
			end
		end
	end
	if F.priority == "Nearest" then
		local hrp = getHRP()
		if hrp then
			table.sort(list, function(a,b)
				local da = a.part and (a.part.Position - hrp.Position).Magnitude or 1e9
				local db = b.part and (b.part.Position - hrp.Position).Magnitude or 1e9
				return da < db
			end)
		end
	else
		table.sort(list, function(a,b) return (a.score or 0) > (b.score or 0) end)
	end
	return list
end

	return list
end
local function guardDist(pos)
	local best = math.huge
	for _, g in ipairs(findGuards()) do
		local d = (g.Position - pos).Magnitude
		if d < best then best = d end
	end
	return best
end
local function agHomePos()
	local b = findBase()
	if b then return b.Position + Vector3.new(0, 3, 0) end
	local hrp = getHRP()
	return hrp and hrp.Position or Vector3.zero
end
local function agLinePos(cfg, fromPos)
	local world = Workspace:FindFirstChild("World") or Workspace:FindFirstChild("__OBJECTS")
	local areas = world and world:FindFirstChild("Areas")
	local sep = areas and areas:FindFirstChild("SeparationLine")
	if sep and sep:IsA("BasePart") then
		local cf = sep.CFrame
		local axis = (Vector3.new(0,1,0)):Cross(sep.Size.X >= sep.Size.Z and cf.RightVector or cf.LookVector)
		local flat = Vector3.new(axis.X, 0, axis.Z)
		if flat.Magnitude > 0.001 then
			local unit = flat.Unit
			local side = ((fromPos - cf.Position):Dot(unit) >= 0) and -unit or unit
			local p = cf.Position + side * (tonumber(cfg.LineOffset) or 8)
			return Vector3.new(p.X, fromPos.Y + 0.5, p.Z)
		end
	end
	return nil
end
local function agOffsetPos(cfg, startPos)
	local t = tostring(cfg.Target or "line")
	local base = startPos
	if t == "sky" then
		base = startPos + Vector3.new(0, tonumber(cfg.Height) or 45, 0)
	elseif t == "line" then
		base = agLinePos(cfg, startPos) or agHomePos()
	else
		base = agHomePos()
	end
	return base + Vector3.new(tonumber(cfg.OffsetX) or 0, tonumber(cfg.Height) or 0, tonumber(cfg.OffsetZ) or 0)
end

local function agRunSequence(startedAt)
	local cfg = AG_CFG
	local hrp = getHRP()
	local hum = getHum()
	if not hrp or not hum then AG.Busy = false AG.Active = false return end
	local startPos = hrp.Position
	local homeish = agOffsetPos(cfg, startPos)
	local wasStand = hum.PlatformStand
	F.status = "Anti Guard is slipping past the guard"
	AG.Busy = true AG.BusySince = os.clock() AG.Active = true
	if cfg.Limp ~= false then pcall(function() hum.PlatformStand = true end) end
	local function waitUntil(at)
		while F.alive and AG.Enabled and F.antiGuard and (os.clock() - startedAt) < at do
			RunService.Heartbeat:Wait()
		end
		return F.alive and AG.Enabled and F.antiGuard
	end
	for _, step in ipairs(cfg.Steps or {}) do
		if type(step) ~= "table" then break end
		if not waitUntil(tonumber(step.At) or 0) then break end
		local dest = (step.To == "start") and startPos or homeish
		pcall(function()
			local h = getHRP()
			if h then h.CFrame = CFrame.new(dest) end
		end)
		F.status = "Anti Guard step " .. tostring(step.To)
	end
	waitUntil(tonumber(cfg.ReleaseAt) or 0.8)
	F.status = "Anti Guard done, rising up"
	pcall(function()
		local h = getHum()
		if h then h.PlatformStand = wasStand end
		local r = getHRP()
		if r then r.CFrame = r.CFrame + Vector3.new(0, 4, 0) end
	end)
	AG.Busy = false AG.Active = false
	F.status = isCarrying() and "Anti Guard done, getting ready" or "Anti Guard done"
end

local function agStartIfNeeded()
	if not AG.Enabled or not F.antiGuard then return end
	if AG.Active or AG.Busy then return end
	if not AG.Carrying then return end
	if (AG.HitArms or 0) > 0 and (os.clock() - (AG.HitArmedAt or 0)) <= 21 then return end
	F.status = "Waiting for Anti Guard to start"
	AG.Busy = true AG.BusySince = os.clock()
	task.spawn(function()
		local ok, err = pcall(agRunSequence, os.clock())
		if not ok then
			warn("[Space Hub] AG", err)
			AG.Busy = false AG.Active = false
			local h = getHum()
			if h then pcall(function() h.PlatformStand = false end) end
		end
	end)
end

-- Grab (Chilli r337): prompt within 14 + CarryFieldEgg fallback
local function grabEgg(entry)
	local hrp = getHRP()
	if not hrp or not entry then return false end
	if AG.Busy or AG.Active then
		F.status = "Waiting for Anti Guard to finish"
		task.wait(0.35)
	end
	if entry.part then
		pcall(function() hrp.CFrame = entry.part.CFrame + Vector3.new(0, 3, 0) end)
		task.wait(0.12)
	end
	local prompt = entry.prompt
	if not prompt and entry.model then
		prompt = entry.model:FindFirstChild("CarryAreaEgg", true)
		if not prompt then
			for _, d in ipairs(entry.model:GetDescendants()) do
				if d:IsA("ProximityPrompt") then prompt = d break end
			end
		end
	end
	-- search SmartPromptPart style within 14
	if not prompt then
		for _, d in ipairs(Workspace:GetDescendants()) do
			if d:IsA("ProximityPrompt") and d.Enabled and d.Name == "CarryAreaEgg" then
				local p = d.Parent
				if p and p:IsA("BasePart") and (p.Position - hrp.Position).Magnitude <= 14 then
					prompt = d
					break
				end
			end
		end
	end
	if prompt then
		firePrompt(prompt)
	end
	-- fallback EggState.CarryFieldEgg(uid)
	if entry.uid and EggState and type(EggState.CarryFieldEgg) == "function" then
		pcall(function() EggState.CarryFieldEgg(entry.uid) end)
	end
	local t0 = os.clock()
	while os.clock() - t0 < 1.5 do
		if isCarrying() then
			if entry.uid then eggCooldown[entry.uid] = os.clock() end
			return true
		end
		task.wait(0.08)
	end
	return isCarrying()
end

-- Safe delivery home + DropFieldEgg
local function deliverHome()
	F.status = "Delivering home"
	-- wait AG if busy
	local t0 = os.clock()
	while (AG.Busy or AG.Active) and os.clock() - t0 < 4 do
		task.wait(0.2)
	end
	if F.safeDelivery then
		for _ = 1, 8 do
			local hrp = getHRP()
			if not hrp then break end
			if guardDist(hrp.Position) > 20 then break end
			task.wait(0.3)
		end
	end
	local base = findBase()
	if base then goTo(base) end
	task.wait(0.25)
	-- Drop
	if EggState and type(EggState.DropFieldEgg) == "function" then
		pcall(function() EggState.DropFieldEgg("PlayerRequest") end)
	end
	invoke("RF/EggWorld/AskPlaceEgg", {})
end

local function stealBest()
	if shouldWaitEnv() then
		task.wait(1.2)
		return false
	end
	F.status = "Scanning eggs"
	local eggs = collectEggs()
	local need = RarityRank[F.minRarity] or 3
	for _, e in ipairs(eggs) do
		if rarityScore(e.rarity) >= need and (e.value or 0) >= F.minStealValue then
			F.status = "Stealing " .. tostring(e.rarity)
			if grabEgg(e) then
				local hum = getHum()
				if hum and F.carrySpeed > 16 then hum.WalkSpeed = F.carrySpeed end
				deliverHome()
				return true
			end
		end
	end
	if #eggs > 0 then
		if grabEgg(eggs[1]) then deliverHome() return true end
	end
	return false
end

local function wearTM() invoke("RF/Treadmill/AskWearStill") end
local function doffTM() invoke("RF/Treadmill/AskDoff") end
local function farmTM(sec)
	local tm = findTreadmill()
	if tm then goTo(tm) end
	wearTM()
	local t0 = os.clock()
	while F.alive and (F.autoFarm or F.autoTreadmill) and os.clock() - t0 < sec do
		local hum = getHum()
		if hum then
			pcall(function()
				hum.WalkSpeed = math.max(hum.WalkSpeed, F.speedVal)
				hum:Move(Vector3.new(1,0,0), true)
			end)
		end
		local hrp = getHRP()
		if hrp and tm and (hrp.Position - tm.Position).Magnitude > 30 then
			goTo(tm)
			if F.stayTreadmill then wearTM() end
		end
		task.wait(0.3)
	end
end

local function runAutoFarm()
	while F.alive and F.autoFarm do
		F.status = "Steal phase"
		stealBest()
		if not F.autoFarm then break end
		if F.autoUpgradeTM then invoke("RF/Treadmill/AskTierRaise") end
		F.status = "Treadmill phase"
		farmTM(24)
		if not F.autoFarm then break end
		F.status = "Restock wait"
		task.wait(1.4)
	end
	F.status = "Idle"
end

local function slider(page, label, order, min, max, default, cb)
	local row = Instance.new("Frame")
	row.Size = UDim2.new(1,-8,0,42)
	row.BackgroundColor3 = Color3.fromRGB(18,20,38)
	row.BorderSizePixel = 0
	row.LayoutOrder = order
	row.Parent = page
	corner(row, 6)
	local val = default
	local t = Instance.new("TextLabel")
	t.Size = UDim2.new(1,-12,0,16)
	t.Position = UDim2.fromOffset(8,3)
	t.BackgroundTransparency = 1
	t.Text = label .. ": " .. tostring(val)
	t.TextColor3 = COLORS.text
	t.Font = Enum.Font.Gotham
	t.TextSize = 11
	t.TextXAlignment = Enum.TextXAlignment.Left
	t.Parent = row
	local bar = Instance.new("Frame")
	bar.Size = UDim2.new(1,-16,0,7)
	bar.Position = UDim2.fromOffset(8,26)
	bar.BackgroundColor3 = Color3.fromRGB(30,35,60)
	bar.BorderSizePixel = 0
	bar.Parent = row
	corner(bar, 4)
	local fill = Instance.new("Frame")
	fill.Size = UDim2.new((val-min)/math.max(max-min,1),0,1,0)
	fill.BackgroundColor3 = COLORS.accent
	fill.BorderSizePixel = 0
	fill.Parent = bar
	corner(fill, 4)
	local dragging = false
	local function update(x)
		local rel = math.clamp((x - bar.AbsolutePosition.X) / math.max(bar.AbsoluteSize.X, 1), 0, 1)
		val = math.floor(min + (max - min) * rel + 0.5)
		fill.Size = UDim2.new(rel,0,1,0)
		t.Text = label .. ": " .. tostring(val)
		if cb then pcall(cb, val) end
	end
	bar.InputBegan:Connect(function(input)
		if input.UserInputType == Enum.UserInputType.MouseButton1 or input.UserInputType == Enum.UserInputType.Touch then
			dragging = true update(input.Position.X)
		end
	end)
	UserInputService.InputEnded:Connect(function(input)
		if input.UserInputType == Enum.UserInputType.MouseButton1 or input.UserInputType == Enum.UserInputType.Touch then
			dragging = false
		end
	end)
	UserInputService.InputChanged:Connect(function(input)
		if dragging and (input.UserInputType == Enum.UserInputType.MouseMovement or input.UserInputType == Enum.UserInputType.Touch) then
			update(input.Position.X)
		end
	end)
end
