local musicSound = Instance.new("Sound")
musicSound.Name = "NovaHub_Beethoven_FurElise"
musicSound.Looped = true
musicSound.Volume = 0.5
musicSound.Parent = SoundService
local FUR_IDS = {
	"rbxassetid://1848056136",
	"rbxassetid://1837626326",
	"rbxassetid://450051032",
	"rbxassetid://1836283178",
}
local function startMusic()
	task.spawn(function()
		for _, id in ipairs(FUR_IDS) do
			if getgenv().KEY_PASSED then return end
			musicSound.SoundId = id
			pcall(function() musicSound:Play() end)
			task.wait(0.45)
			if musicSound.IsPlaying or (musicSound.TimeLength or 0) > 0 then return end
		end
	end)
end
local function stopMusic()
	pcall(function()
		if musicSound then
			musicSound:Stop()
			musicSound:Destroy()
			musicSound = nil
		end
		for _, s in ipairs(SoundService:GetChildren()) do
			if s:IsA("Sound") and (s.Name:find("FurElise") or s.Name:find("Beethoven") or s.Name:find("NovaHub_")) then
				s:Stop()
				s:Destroy()
			end
		end
	end)
end

local function corner(p, r)
	local c = Instance.new("UICorner")
	c.CornerRadius = UDim.new(0, r or 12)
	c.Parent = p
	return c
end
local function stroke(p, col, th)
	local s = Instance.new("UIStroke")
	s.Color = col
	s.Thickness = th or 1.2
	s.ApplyStrokeMode = Enum.ApplyStrokeMode.Border
	s.Parent = p
	return s
end
local function label(parent, props)
	local l = Instance.new("TextLabel")
	l.BackgroundTransparency = 1
	for k, v in pairs(props) do l[k] = v end
	l.Parent = parent
	return l
end
local function glowStroke(parent, col, thickness)
	local g = Instance.new("UIStroke")
	g.Name = "Glow"
	g.Color = col
	g.Thickness = thickness or 8
	g.Transparency = 0.65
	g.Parent = parent
	return g
end
local function pulseGlow(glow, accentFn)
	task.spawn(function()
		local a = 0
		while glow and glow.Parent do
			a = a + 0.045
			local c = accentFn and accentFn() or glow.Color
			glow.Color = c
			glow.Transparency = 0.48 + 0.28 * (0.5 + 0.5 * math.sin(a))
			task.wait(0.03)
		end
	end)
end

local function isProteinOrConsumable(name)
    local n = string.lower(tostring(name or ""))
    return string.find(n, "protein")
        or string.find(n, "shake")
        or string.find(n, "potion")
        or string.find(n, "boost")
        or string.find(n, "food")
        or string.find(n, "drink")
end

local function getEquippedTool()
    local char = LocalPlayer.Character
    if not char then
        return
    end
    local tool = char:FindFirstChildOfClass("Tool")
    if tool then
        local n = string.lower(tool.Name)
        if not string.find(n, "punch") and not isProteinOrConsumable(tool.Name) then
            return tool
        end
    end
    local backpack = LocalPlayer:FindFirstChild("Backpack")
    if backpack then
        for _, item in ipairs(backpack:GetChildren()) do
            if item:IsA("Tool") then
                local n = string.lower(item.Name)
                if not string.find(n, "punch") and not isProteinOrConsumable(item.Name) then
                    local hum = char:FindFirstChildOfClass("Humanoid")
                    if hum then
                        hum:EquipTool(item)
                    end
                    return item
                end
            end
        end
    end
    return nil
end

local function unequipProtein()
    pcall(function()
        local char = LocalPlayer.Character
        if not char then

        local tool = char:FindFirstChildOfClass("Tool")

            local hum = char:FindFirstChildOfClass("Humanoid")
            if hum then
                hum:UnequipTools()
            end
        end
    end)
end

local function getRockCFrame(rock)
    if not rock then
        return nil
    end
    if rock:IsA("Model") then
        local primary = rock.PrimaryPart or rock:FindFirstChildWhichIsA("BasePart")
        if primary then
            return primary.CFrame
        end
    elseif rock:IsA("BasePart") then
        return rock.CFrame
    end
    for _, part in ipairs(rock:GetDescendants()) do
        if part:IsA("BasePart") then
            return part.CFrame
        end
    end
    return nil
end

local function equipExerciseTool(toolName)
    local char = LocalPlayer.Character
    local backpack = LocalPlayer:FindFirstChild("Backpack")
    local hum = char and char:FindFirstChildOfClass("Humanoid")
    if not char or not hum then
        return false
    end
    if char:FindFirstChild(toolName) then
        return true
    end
    if backpack then
        local tool = backpack:FindFirstChild(toolName)
        if tool then
            hum:EquipTool(tool)
            return true
        end
        -- giveTool fallback
        local give = RemoteEvents and RemoteEvents:FindFirstChild("giveTool")
        if give then
            pcall(function()
                give:FireServer(toolName)
            end)
            task.wait(0.1)
            tool = backpack:FindFirstChild(toolName) or char:FindFirstChild(toolName)
            if tool and tool.Parent == backpack then
                hum:EquipTool(tool)
            end
            return tool ~= nil
        end
    end
    return false
end

local function getNearestMachineCFrame()
    local folder = workspace:FindFirstChild("machinesFolder")
    local myChar = LocalPlayer.Character
    local myRoot = myChar and myChar:FindFirstChild("HumanoidRootPart")
    if not folder or not myRoot then
        return nil
    end
    local best, bestDist = nil, math.huge
    for _, obj in ipairs(folder:GetDescendants()) do
        if obj:IsA("BasePart") then
            local d = (obj.Position - myRoot.Position).Magnitude
            if d < bestDist then
                bestDist = d
                best = obj
            end
        end
    end
    if best then
        return best.CFrame * CFrame.new(0, 0, 3)
    end
    return nil
end

local function packGetRequiredStrength()
    local ls = LocalPlayer:FindFirstChild("leaderstats")
    local reb = ls and ls:FindFirstChild("Rebirths")
    local v = reb and tonumber(reb.Value) or 0
    return 5000 + v * 2550
end

local function packGetStrength()
    local ls = LocalPlayer:FindFirstChild("leaderstats")
    local s = ls and ls:FindFirstChild("Strength")
    return s and tonumber(s.Value) or 0
end

local function packGetPetsByName(name)
    local pets = {}
    local folder = packPetsFolder()
    if not folder then
        return pets
    end
    local target = string.lower(name)
    for _, sub in ipairs(folder:GetChildren()) do
        if sub:IsA("Folder") then
            for _, pet in ipairs(sub:GetChildren()) do
                if string.lower(pet.Name) == target then
                    pets[#pets + 1] = pet
                end
            end
        end
    end
    return pets
end

local function packUnequipAll()
    local equip = packEquipRemote()
    local folder = packPetsFolder()
    if not equip or not folder then
        return
    end
    for _, sub in ipairs(folder:GetChildren()) do
        if sub:IsA("Folder") then
            for _, pet in ipairs(sub:GetChildren()) do
                pcall(function()
                    equip:FireServer("unequipPet", pet)
                end)
            end
        end
    end
    task.wait(0.15)
end

local function packEquipAmount(petName, amount)
    local equip = packEquipRemote()
    if not equip then
        return 0
    end
    local pets = packGetPetsByName(petName)
    local equipped = 0
    for _, pet in ipairs(pets) do
        if equipped >= amount then
            break
        end
        pcall(function()
            equip:FireServer("equipPet", pet)
        end)
        equipped = equipped + 1
        task.wait(0.04)
    end
    return equipped
end

local function packEquipTraining()
    packUnequipAll()
    return packEquipAmount(TRAINING_PET, PET_STACK)
end

local function packEquipRebirthPets()
    packUnequipAll()
    return packEquipAmount(REBIRTH_PET, PET_STACK)
end

local function packTrainFast()
    local ev = packMuscleEv()
    if not ev then
        return
    end
    for _ = 1, 30 do
        pcall(function()
            ev:FireServer("rep")
        end)
    end
end

local function packTrainRebirth()
    local ev = packMuscleEv()
    if not ev then
        return
    end
    for _ = 1, 10 do
        pcall(function()
            ev:FireServer("rep")
        end)
    end
end

local function packDoRebirth()
    local required = packGetRequiredStrength()
    if packGetStrength() < required then
        return false
    end
    packEquipRebirthPets()
    task.wait(0.25)
    local ls = LocalPlayer:FindFirstChild("leaderstats")
    local rebStat = ls and ls:FindFirstChild("Rebirths")
    local old = rebStat and rebStat.Value or 0
    local started = tick()
    local remote = ReplicatedStorage:FindFirstChild("rEvents")
        and ReplicatedStorage.rEvents:FindFirstChild("rebirthRemote")
    while getgenv().packFastRebirth and rebStat and rebStat.Value <= old and tick() - started < 8 do
        pcall(function()
            if remote then
                remote:InvokeServer("rebirthRequest")
            end
        end)
        task.wait(0.05)
    end
    return rebStat and rebStat.Value > old
end

local function startPackFastFarm()
    getgenv().packFarmGen = (getgenv().packFarmGen or 0) + 1
    local gen = getgenv().packFarmGen
    task.spawn(function()
        packEquipTraining()
        while getgenv().packFastFarm and gen == getgenv().packFarmGen do
            packTrainFast()
            task.wait(0.01)
        end
    end)
end

local function startPackFastRebirth()
    getgenv().packRebGen = (getgenv().packRebGen or 0) + 1
    local gen = getgenv().packRebGen
    task.spawn(function()
        packEquipTraining()
        while getgenv().packFastRebirth and gen == getgenv().packRebGen do
            local required = packGetRequiredStrength()
            if packGetStrength() < required then
                packTrainRebirth()
                task.wait(0.02)
            else
                local ok = packDoRebirth()
                if not getgenv().packFastRebirth or gen ~= getgenv().packRebGen then
                    break
                end
                task.wait(ok and 0.15 or 0.25)
                packEquipTraining()
            end
        end
    end)
end

local function buyPetShop(petName)
    if not petName then
        return false
    end
    local ok = false
    pcall(function()
        local remote = ReplicatedStorage:FindFirstChild("rEvents")
            and ReplicatedStorage.rEvents:FindFirstChild("cPetShopRemote")
        if not remote then
            remote = ReplicatedStorage:FindFirstChild("cPetShopRemote")
        end
        local petObj = nil
        local shared = ReplicatedStorage:FindFirstChild("shared")
        local runtime = shared and shared:FindFirstChild("runtime")
        local folder = runtime and runtime:FindFirstChild("cPetShopFolder")
        if folder then
            petObj = folder:FindFirstChild(petName)
        end
        if not petObj then
            folder = ReplicatedStorage:FindFirstChild("cPetShopFolder")
            petObj = folder and folder:FindFirstChild(petName)
        end
        if remote and petObj then
            if remote:IsA("RemoteFunction") then
                remote:InvokeServer(petObj)
            else
                remote:FireServer(petObj)
            end
            ok = true
            return
        end
        -- fallback: DAN public style root remote + folder
        if not remote then
            remote = ReplicatedStorage:FindFirstChild("cPetShopRemote")
        end
        if remote and petObj and remote:IsA("RemoteFunction") then
            remote:InvokeServer(petObj)
            ok = true
        end
    end)
    return ok
end

local playerNameSet = {}
local function refreshPlayerNames()
    playerNameSet = {}
    for _, plr in ipairs(Players:GetPlayers()) do
        playerNameSet[plr.Name] = true
        if plr.Character then
            playerNameSet[plr.Character.Name] = true
        end
    end

local function getBossRoot(model)
    if not model then
        return nil
    end
    return model:FindFirstChild("HumanoidRootPart")
        or model:FindFirstChild("Torso")
        or model:FindFirstChild("UpperTorso")
        or model:FindFirstChild("Root")
        or model.PrimaryPart
        or model:FindFirstChildWhichIsA("BasePart", true)
end

local function isBossPet(model)
	if not model then return true end
	local n = string.lower(model.Name)
	if string.find(n, "pet") or string.find(n, "evolved") then
		return true
	end
	for _, d in ipairs(model:GetDescendants()) do
		if d:IsA("TextLabel") or d:IsA("TextButton") then
			local t = string.lower(tostring(d.Text or ""))
			if string.find(t, "pet") or string.find(t, "evolved") then
				return true
			end
		elseif d:IsA("BillboardGui") then
			for _, c in ipairs(d:GetDescendants()) do
				if c:IsA("TextLabel") or c:IsA("TextButton") then
					local t = string.lower(tostring(c.Text or ""))
					if string.find(t, "pet") or string.find(t, "evolved") then
						return true
					end
				end
			end
		end
	end
	local p = model.Parent
	for _ = 1, 8 do
		if not p then break end
		if string.find(string.lower(p.Name), "pet") then
			return true
		end
		p = p.Parent
	end
	return false
end

local function isLivingBoss(model)
	if not model or not model:IsA("Model") or not model.Parent then
		return false
	end
	if model == LocalPlayer.Character then
		return false
	end
	if Players:GetPlayerFromCharacter(model) then
		return false
	end
	if playerNameSet[model.Name] then
		return false
	end
	if isBossPet(model) then
		return false
	end
	local hum = model:FindFirstChildOfClass("Humanoid")
	if not hum or hum.Health <= 0 then
		return false
	end
	local root = getBossRoot(model)
	if not root or not root.Parent then
		return false
	end
	if root.Position.Y < -50 then
		return false
	end
	return true, hum, root
end

end

local function rarityMatches(name)
	local rarity = getgenv().selectedBossRarity or "Any"
	if rarity == "Any" then
		return true
	end
	local n, r = string.lower(name), string.lower(rarity)
	if r == "legendary" or r == "legend" then
		return string.find(n, "legend") ~= nil
	end
	if r == "mythic" or r == "myth" then
		return string.find(n, "myth") ~= nil
	end
	if r == "rainbow" then
		return string.find(n, "rainbow") ~= nil or string.find(n, "admin") ~= nil
	end
	return string.find(n, r) ~= nil
end

-- Real boss fights always have this hitbox. Pets do not.
local function findBossHitbox(boss)
	if boss then
		for _, n in ipairs({"BossDamageHitbox", "BossHitbox", "DamageHitbox"}) do
			local hb = boss:FindFirstChild(n, true)
			if hb and hb:IsA("BasePart") then
				return hb
			end
		end
	end
	for _, n in ipairs({"BossDamageHitbox", "BossHitbox"}) do
		local hb = workspace:FindFirstChild(n, true)
		if hb and hb:IsA("BasePart") then
			return hb
		end
	end
	return nil
end

local function modelNearPart(part, maxDist)
	if not part then
		return nil
	end
	local best, bestD = nil, maxDist or 80
	for _, obj in ipairs(workspace:GetDescendants()) do
		if obj:IsA("Humanoid") and obj.Health > 0 then
			local model = obj.Parent
			if model and model:IsA("Model") and isLivingBoss(model) then
				local root = getBossRoot(model)
				if root then
					local d = (root.Position - part.Position).Magnitude
					if d < bestD then
						bestD = d
						best = model
					end
				end
			end
		end
	end
	return best
end

local function findBossModel()
	refreshPlayerNames()

	if getgenv().lockedBoss and getgenv().lockedBoss.Parent and isLivingBoss(getgenv().lockedBoss) then
		local hb = findBossHitbox(getgenv().lockedBoss)
		-- keep lock only if real boss hitbox still exists nearby or on model
		if hb or findBossHitbox() then
			return getgenv().lockedBoss
		end
	end
	getgenv().lockedBoss = nil

	-- 1) hitbox first = only when a real boss fight is active
	local hb = findBossHitbox()
	if hb then
		local model = hb:FindFirstAncestorOfClass("Model")
		if model and isLivingBoss(model) then
			getgenv().lockedBoss = model
			return model
		end
		local near = modelNearPart(hb, 120)
		if near then
			getgenv().lockedBoss = near
			return near
		end
	end

	-- 2) named boss models (still require not pet + high hp)
	for _, name in ipairs({
		"Boss", "BossArena", "CommonBoss", "RareBoss", "EpicBoss",
		"LegendaryBoss", "MythicBoss", "RainbowBoss", "CurrentBoss"
	}) do
		local m = workspace:FindFirstChild(name, true)
		if m then
			if not m:IsA("Model") then
				m = m:FindFirstAncestorOfClass("Model") or m
			end
			if m:IsA("Model") and isLivingBoss(m) then
				local hum = m:FindFirstChildOfClass("Humanoid")
				if hum and hum.MaxHealth >= 1000 then
					getgenv().lockedBoss = m
					return m
				end
			end
		end
	end

	-- 3) folders that only exist for boss content
	for _, folderName in ipairs({"Bosses", "BossArena", "Boss", "Raids", "Raid"}) do
		local folder = workspace:FindFirstChild(folderName)
		if folder then
			for _, obj in ipairs(folder:GetDescendants()) do
				if obj:IsA("Model") and isLivingBoss(obj) then
					local hum = obj:FindFirstChildOfClass("Humanoid")
					if hum and hum.MaxHealth >= 1000 then
						getgenv().lockedBoss = obj
						return obj
					end
				end
			end
		end
	end

	-- no real boss up (lobby / pets only)
	return nil
end

local function bossPunch()
	local punch = equipPunch()
	local ev = LocalPlayer:FindFirstChild("muscleEvent") or MuscleEvent
	if punch then
		pcall(function()
			punch:Activate()
		end)
	end
	if ev then
		ev:FireServer("punch", "leftHand")
		ev:FireServer("punch", "rightHand")
		ev:FireServer("punch")
	end
end

local function bossPunchFast()
	-- keep Punch equipped forever while fighting; low-lag remote spam
	local char = LocalPlayer.Character
	local punch = char and char:FindFirstChild("Punch")
	if not punch then
		punch = equipPunch()
	end
	local ev = LocalPlayer:FindFirstChild("muscleEvent") or MuscleEvent
	if punch then
		pcall(function() punch:Activate() end)
	end
	if ev then
		for _ = 1, 8 do
			ev:FireServer("punch", "leftHand")
			ev:FireServer("punch", "rightHand")
			ev:FireServer("punch")
		end
	end
end

local function bossHeavy()
    local ev = LocalPlayer:FindFirstChild("muscleEvent") or MuscleEvent
    if not ev then
        return
    end
    pcall(function()
        local bag = LocalPlayer:FindFirstChild("Backpack")
        local char = LocalPlayer.Character
        local hum = char and char:FindFirstChildOfClass("Humanoid")
        if bag and hum then
            local stomp = bag:FindFirstChild("Stomp") or char:FindFirstChild("Stomp")
            local slam = bag:FindFirstChild("Ground Slam") or char:FindFirstChild("Ground Slam")

                hum:EquipTool(stomp)

        ev:FireServer("slam")
        equipPunch()
    end)
end

local function ensureBossAntiKnock(myRoot)
    if not myRoot then
        return
    end
    local bv = myRoot:FindFirstChild("DAN_BOSS_BV")
    if not bv then
        bv = Instance.new("BodyVelocity")
        bv.Name = "DAN_BOSS_BV"
        bv.MaxForce = Vector3.new(1e5, 1e5, 1e5)
        bv.P = 1250
        bv.Parent = myRoot
    end
    bv.Velocity = Vector3.new(0, 0, 0)
end

local function clearBossAntiKnock(myRoot)
    if myRoot then
        local bv = myRoot:FindFirstChild("DAN_BOSS_BV")
        if bv then
            bv:Destroy()
        end
    end
end

local function bossStick(myRoot, bossRoot)
	local h = tonumber(getgenv().bossHeight) or 68
	local range = tonumber(getgenv().bossRange) or 6
	-- stay above boss and follow
	local pos = bossRoot.Position + Vector3.new(0, h, 0)
	myRoot.CFrame = CFrame.new(pos, bossRoot.Position)
	pcall(function()
		myRoot.AssemblyLinearVelocity = Vector3.zero
	end)
	ensureBossAntiKnock(myRoot)
end

local function bossDodge(myRoot, bossRoot)
    local h = tonumber(getgenv().bossHeight) or 10
    local range = tonumber(getgenv().bossRange) or 8
    local side = Vector3.new(1, 0, 0)
    local away = (myRoot.Position - bossRoot.Position)
    if away.Magnitude > 0.1 then
        side = away:Cross(Vector3.new(0, 1, 0))
        if side.Magnitude > 0.1 then
            side = side.Unit
        end
    end
    local dir = (math.random(0, 1) == 0) and 1 or -1
    local pos = bossRoot.Position + side * (range + 6) * dir + Vector3.new(0, h, 0)
    myRoot.CFrame = CFrame.new(pos, bossRoot.Position)
    ensureBossAntiKnock(myRoot)
end

-- Cache boss body parts (no workspace scan every frame = less lag)
local function refreshBossParts(bossModel, hitbox)
	local targets = {}
	if hitbox then
		targets[#targets + 1] = hitbox
	end
	if bossModel then
		for _, p in ipairs(bossModel:GetDescendants()) do
			if p:IsA("BasePart") then
				targets[#targets + 1] = p
			end
		end
	end
	getgenv()._bossParts = targets
	return targets
end

local function ensureHugeHitbox(myRoot)
	if not myRoot then return end
	local box = myRoot:FindFirstChild("DAN_HUGE_HB")
	if not box then
		box = Instance.new("Part")
		box.Name = "DAN_HUGE_HB"
		box.Anchored = false
		box.CanCollide = false
		box.Massless = true
		box.Transparency = 1
		box.Size = Vector3.new(80, 80, 80)
		box.Parent = myRoot
		local w = Instance.new("WeldConstraint")
		w.Part0 = myRoot
		w.Part1 = box
		w.Parent = box
	end
	box.CFrame = myRoot.CFrame
	return box
end

local function hitBossCached(myRoot)
	local targets = getgenv()._bossParts
	if not myRoot or not targets then return end
	local huge = ensureHugeHitbox(myRoot)
	if firetouchinterest then
		for i = 1, #targets do
			local part = targets[i]
			if part and part.Parent then
				pcall(firetouchinterest, part, myRoot, 0)
				pcall(firetouchinterest, part, myRoot, 1)
				if huge then
					pcall(firetouchinterest, part, huge, 0)
					pcall(firetouchinterest, part, huge, 1)
				end
			end
		end
	end
end

local function bossHitCycle(myRoot, targetPart, boss, hitbox)
	if not myRoot or not targetPart then return end
	equipPunch()
	-- always follow boss at height
	bossStick(myRoot, targetPart)
	hitBossCached(myRoot)
	bossPunchFast()
	-- quick touch then stay high
	myRoot.CFrame = CFrame.new(targetPart.Position + Vector3.new(0, 6, 0), targetPart.Position)
	hitBossCached(myRoot)
	bossPunchFast()
	bossStick(myRoot, targetPart)
	hitBossCached(myRoot)
	bossPunchFast()
end

-- Hide game boss reward / menu popup (the panel in screenshot)
getgenv().hideBossUI = true
local function hideBossPopup()
    if getgenv().hideBossUI == false then
        return
    end
    local pg = LocalPlayer:FindFirstChild("PlayerGui")
    if not pg then
        return
    end
    for _, gui in ipairs(pg:GetChildren()) do
        local n = string.lower(gui.Name)
        if string.find(n, "boss") or string.find(n, "reward") or string.find(n, "chest") then
            pcall(function()
                if gui:IsA("ScreenGui") or gui:IsA("GuiMain") then
                    gui.Enabled = false
                end
            end)
        end
        for _, d in ipairs(gui:GetDescendants()) do
            if d:IsA("TextLabel") or d:IsA("TextButton") then
                local t = string.lower(d.Text or "")
                if string.find(t, "click for rewards") or string.find(t, "chest rewards")
                    or string.find(t, "common boss") or string.find(t, "spawn rate") then
                    local root = d
                    for _ = 1, 8 do
                        if root.Parent and root.Parent ~= pg then
                            root = root.Parent
                        else
                            break
                        end
                    end
                    pcall(function()
                        if root:IsA("ScreenGui") then
                            root.Enabled = false
                        elseif root:IsA("Frame") or root:IsA("ImageLabel") then
                            root.Visible = false
                        end
                    end)
                end
            end
        end
    end
end

        if Value then
            task.spawn(function()
                local lastHealth = nil
                local tickN = 0
                local cachedBoss, cachedHitbox, cachedRoot = nil, nil, nil
                local lastScan, lastPopup, lastEquip = 0, 0, 0
                while getgenv().autoKillBoss do
                    pcall(function()
                        local now = tick()
                        -- scan boss less often (less lag)
                        if now - lastScan > 0.35 or not cachedBoss or not cachedBoss.Parent then
                            lastScan = now
                            cachedBoss = findBossModel()
                            cachedHitbox = findBossHitbox(cachedBoss)
                            if cachedBoss and isLivingBoss(cachedBoss) then
                                cachedRoot = getBossRoot(cachedBoss)
                                if now - lastScan > 0.34 then
                                    refreshBossParts(cachedBoss, cachedHitbox)
                                end
                            else
                                cachedBoss, cachedHitbox, cachedRoot = nil, cachedHitbox, nil
                            end
                        end

                        if now - lastPopup > 2 then
                            lastPopup = now
                            hideBossPopup()
                        end

                        local boss, hitbox = cachedBoss, cachedHitbox
                        if not boss and not hitbox then
                            clearBossTarget()
                            return
                        end
                        if boss and not isLivingBoss(boss) then
                            clearBossTarget()
                            cachedBoss = nil
                            return
                        end

                        local char = LocalPlayer.Character
                        local myRoot = char and char:FindFirstChild("HumanoidRootPart")
                        local myHum = char and char:FindFirstChildOfClass("Humanoid")
                        if not myRoot then
                            return
                        end

                        local targetPart = cachedRoot or (boss and getBossRoot(boss)) or hitbox
                        if not targetPart or not targetPart.Parent then
                            clearBossTarget()
                            cachedBoss = nil
                            return
                        end
                        if targetPart.Position.Y < -50 then
                            clearBossTarget()
                            cachedBoss = nil
                            return
                        end

                        -- keep Punch equipped the whole fight
                        local held = char and char:FindFirstChild("Punch")
                        if not held then
                            equipPunch()
                        end

                        bossStick(myRoot, targetPart)

                        -- only touch hitbox (not every boss part)
                        if hitbox and firetouchinterest then
                            pcall(firetouchinterest, hitbox, myRoot, 0)
                            pcall(firetouchinterest, hitbox, myRoot, 1)
                            local huge = myRoot:FindFirstChild("DAN_HUGE_HB")

                            else
                                ensureHugeHitbox(myRoot)
                            end
                        else
                            hitBossCached(myRoot)
                        end

                        -- fast punch (optimized)
                        bossPunchFast()

                        if myHum and lastHealth and myHum.Health < lastHealth - 2 then
                            bossDodge(myRoot, targetPart)
                            bossStick(myRoot, targetPart)
                        end
                        if myHum then
                            lastHealth = myHum.Health
                        end
                    end)
                    -- tight loop for punch rate, scans already throttled
                    task.wait(0.03)
                end
            end)
        else
            clearBossTarget()
            local myRoot = LocalPlayer.Character and LocalPlayer.Character:FindFirstChild("HumanoidRootPart")
            if myRoot then

        end

-- STAT TRACKING (local player gains)

local function readLocalStats()
    local stats = {
        Strength = 0,
        Durability = 0,
        Rebirths = 0,
        Kills = 0,
        Brawls = 0
    }
    pcall(function()
        local ls = LocalPlayer:FindFirstChild("leaderstats")
        if ls then
            for _, v in ipairs(ls:GetChildren()) do
                local n = string.lower(v.Name)
                local val = tonumber(v.Value) or 0
                if string.find(n, "stren") then
                    stats.Strength = val
                elseif string.find(n, "dura") then
                    stats.Durability = val
                elseif string.find(n, "rebirth") then
                    stats.Rebirths = val
                elseif string.find(n, "kill") then
                    stats.Kills = val
                elseif string.find(n, "brawl") then
                    stats.Brawls = val
                end
            end
        end
        -- Muscle Legends often stores more under player folders
        for _, folderName in ipairs({"MuscleStats", "stats", "Stats", "playerStats"}) do
            local f = LocalPlayer:FindFirstChild(folderName)
            if f then
                for _, v in ipairs(f:GetDescendants()) do
                    if v:IsA("NumberValue") or v:IsA("IntValue") then
                        local n = string.lower(v.Name)
                        local val = tonumber(v.Value) or 0
                        if string.find(n, "stren") and val > stats.Strength then
                            stats.Strength = val
                        elseif string.find(n, "dura") and val > stats.Durability then
                            stats.Durability = val
                        elseif string.find(n, "rebirth") and val > stats.Rebirths then
                            stats.Rebirths = val
                        end
                    end
                end
            end
        end
    end)
    return stats
end

local function formatNum(n)
    n = tonumber(n) or 0
    local abs = math.abs(n)
    if abs >= 1e12 then
        return string.format("%.2fT", n / 1e12)
    elseif abs >= 1e9 then
        return string.format("%.2fB", n / 1e9)
    elseif abs >= 1e6 then
        return string.format("%.2fM", n / 1e6)
    elseif abs >= 1e3 then
        return string.format("%.2fK", n / 1e3)
    end
    return string.format("%.0f", n)
end

local function ratesFor(delta, seconds)
    if seconds <= 0 then
        return 0, 0, 0, 0, 0, 0
    end
    local perSec = delta / seconds
    return perSec * 60, perSec * 3600, perSec * 86400, perSec * 604800, perSec * 2592000, perSec * 31536000
end

local function readPlayerStats(player)
    local stats = {
        Strength = nil,
        Durability = nil,
        Agility = nil,
        Rebirths = nil,
        Kills = nil,
        Brawls = nil
    }
    if not player then
        return stats
    end
    pcall(function()
        local function applyName(n, val)
            local low = string.lower(tostring(n))
            if string.find(low, "stren") then
                stats.Strength = val
            elseif string.find(low, "dura") then
                stats.Durability = val
            elseif string.find(low, "agil") or string.find(low, "speed") then
                stats.Agility = val
            elseif string.find(low, "rebirth") then
                stats.Rebirths = val
            elseif string.find(low, "kill") then
                stats.Kills = val
            elseif string.find(low, "brawl") then
                stats.Brawls = val
            else
                stats[tostring(n)] = val
            end
        end
        local ls = player:FindFirstChild("leaderstats")
        if ls then
            for _, v in ipairs(ls:GetChildren()) do
                if v:IsA("NumberValue") or v:IsA("IntValue") or v:IsA("StringValue") then
                    applyName(v.Name, v.Value)
                end
            end
        end
        for _, folderName in ipairs({"MuscleStats", "stats", "Stats", "playerStats", "Data", "data"}) do
            local f = player:FindFirstChild(folderName)
            if f then
                for _, v in ipairs(f:GetDescendants()) do
                    if v:IsA("NumberValue") or v:IsA("IntValue") then
                        applyName(v.Name, v.Value)
                    end
                end
            end
        end
        -- common alternate attribute names
        for _, attr in ipairs({"Durability", "durability", "Dura", "dura", "Strength", "strength", "Agility", "agility"}) do
            local ok, val = pcall(function()
                return player:GetAttribute(attr)
            end)
            if ok and val ~= nil then
                applyName(attr, val)
            end
        end
    end)
    return stats
end

        end

        names[1] = "(no other players)"
    end
    return names
end

            lines[#lines + 1] = "No public stats found (may be hidden server-side)"
        end

            lines[#lines + 1] = "Durability: (not visible on this player)"
        end

        end)

local function punchTarget(plr)
    if not plr or not plr.Character then
        return
    end
    local targetRoot = plr.Character:FindFirstChild("HumanoidRootPart")
    local myChar = LocalPlayer.Character
    local myRoot = myChar and myChar:FindFirstChild("HumanoidRootPart")
    if not targetRoot or not myRoot then

    myRoot.CFrame = targetRoot.CFrame * CFrame.new(0, 0, 3)
    local punch = equipPunch()
    if punch then
        punch:Activate()
    end
    if MuscleEvent then
        MuscleEvent:FireServer("punch")
        MuscleEvent:FireServer("rep")
    end
end

end
