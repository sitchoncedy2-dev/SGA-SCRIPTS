local function notify(t, m)
	pcall(function()
		StarterGui:SetCore("SendNotification", {
			Title = tostring(t),
			Text = tostring(m),
			Duration = 4
		})
	end)
end

local function Fire(name, a, b, c)
	if not Network then return end
	local r = Network:FindFirstChild(name)
	if not r then return end
	pcall(function()
		if a ~= nil and b ~= nil and c ~= nil then
			r:FireServer(a, b, c)
		elseif a ~= nil and b ~= nil then
			r:FireServer(a, b)
		elseif a ~= nil then
			r:FireServer(a)
		else
			r:FireServer()
		end
	end)
end

local function Invoke(name, a, b, c)
	if not Network then return end
	local r = Network:FindFirstChild(name)
	if not r then return end
	pcall(function()
		if a ~= nil and b ~= nil and c ~= nil then
			r:InvokeServer(a, b, c)
		elseif a ~= nil and b ~= nil then
			r:InvokeServer(a, b)
		elseif a ~= nil then
			r:InvokeServer(a)
		else
			r:InvokeServer()
		end
	end)
end

-- FARM: Breakables_PlayerDealDamage
local function doFarm()
	if not Things then return end
	local folder = Things:FindFirstChild("Breakables")
	if not folder then return end
	local hrp = getHRP()
	if not hrp then return end
	local best = nil
	local bestD = 999999
	local kids = folder:GetChildren()
	local i
	for i = 1, #kids do
		local br = kids[i]
		local part = br:FindFirstChildWhichIsA("MeshPart")
		if not part then
			part = br:FindFirstChildWhichIsA("BasePart")
		end
		if part then
			local d = (part.Position - hrp.Position).Magnitude
			if d < bestD then
				bestD = d
				best = br
			end
		end
	end
	if best then
		if bestD > 90 then
			local part = best:FindFirstChildWhichIsA("MeshPart") or best:FindFirstChildWhichIsA("BasePart")
			if part then
				pcall(function()
					hrp.CFrame = CFrame.new(part.Position + Vector3.new(0, 6, 0))
				end)
			end
		end
		Fire("Breakables_PlayerDealDamage", best.Name)
	end
end

-- ORBS: "Orbs: Collect"
local function doOrbs()
	if not Things then return end
	local folder = Things:FindFirstChild("Orbs")
	if not folder then return end
	local ids = {}
	local kids = folder:GetChildren()
	local i
	for i = 1, #kids do
		local orb = kids[i]
		local id = tonumber(orb.Name)
		if id then
			table.insert(ids, id)
			pcall(function() orb:Destroy() end)
		end
	end
	if #ids > 0 then
		Fire("Orbs: Collect", ids)
	end
end

-- LOOTBAGS
local function doLoot()
	if not Things then return end
	local folder = Things:FindFirstChild("Lootbags")
	if not folder then return end
	local bags = {}
	local has = false
	local kids = folder:GetChildren()
	local i
	for i = 1, #kids do
		local bag = kids[i]
		bags[bag.Name] = bag.Name
		has = true
		pcall(function() bag:Destroy() end)
	end
	if has then
		Fire("Lootbags_Claim", bags)
	end
end

local function doCollect()
	doOrbs()
	doLoot()
end

-- fly
local flyBV = nil
local flyBG = nil
task.spawn(function()
	while S.alive do
		task.wait(0.03)
		if S.fly then
			local r = getHRP()
			if r then
				if not flyBV or flyBV.Parent ~= r then
					if flyBV then pcall(function() flyBV:Destroy() end) end
					if flyBG then pcall(function() flyBG:Destroy() end) end
					flyBV = Instance.new("BodyVelocity")
					flyBV.MaxForce = Vector3.new(9e9, 9e9, 9e9)
					flyBV.Velocity = Vector3.new(0, 0, 0)
					flyBV.Parent = r
					flyBG = Instance.new("BodyGyro")
					flyBG.MaxTorque = Vector3.new(9e9, 9e9, 9e9)
					flyBG.P = 2000
					flyBG.Parent = r
				end
				local cam = Workspace.CurrentCamera
				if cam and flyBV then
					local d = Vector3.new(0, 0, 0)
					if UIS:IsKeyDown(Enum.KeyCode.W) then d = d + cam.CFrame.LookVector end
					if UIS:IsKeyDown(Enum.KeyCode.S) then d = d - cam.CFrame.LookVector end
					if UIS:IsKeyDown(Enum.KeyCode.A) then d = d - cam.CFrame.RightVector end
					if UIS:IsKeyDown(Enum.KeyCode.D) then d = d + cam.CFrame.RightVector end
					if UIS:IsKeyDown(Enum.KeyCode.Space) then d = d + Vector3.new(0, 1, 0) end
					if UIS:IsKeyDown(Enum.KeyCode.LeftControl) then d = d - Vector3.new(0, 1, 0) end
					if d.Magnitude > 0 then
						d = d.Unit * S.flySpd
					end
					flyBV.Velocity = d
					if flyBG then flyBG.CFrame = cam.CFrame end
				end
			end
		else
			if flyBV then pcall(function() flyBV:Destroy() end) flyBV = nil end
			if flyBG then pcall(function() flyBG:Destroy() end) flyBG = nil end
		end
	end
end)

local function mk(class, props, parent)
	local o = Instance.new(class)
	for k, v in pairs(props) do
		o[k] = v
	end
	if parent then o.Parent = parent end
	return o
end

local function corner(o, r)
	local c = Instance.new("UICorner")
	c.CornerRadius = UDim.new(0, r or 4)
	c.Parent = o
end
